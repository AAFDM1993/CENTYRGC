// ── SHA-256 de un string (devuelve hex) ──────────────────────
function sha256_(str) {
  var raw  = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256,
                                      String(str), Utilities.Charset.UTF_8);
  return raw.map(function(b){ return ('0'+(b & 0xFF).toString(16)).slice(-2); }).join('');
}

// ── HMAC-SHA256: firma codigo+rol con TOKEN_SECRET ───────────
function hmacSign_(codigo, rol) {
  var payload = codigo.toLowerCase() + ':' + rol;
  var raw     = Utilities.computeHmacSha256Signature(payload, TOKEN_SECRET,
                                                       Utilities.Charset.UTF_8);
  var sig     = raw.map(function(b){ return ('0'+(b & 0xFF).toString(16)).slice(-2); }).join('');
  // token = base64( codigo:rol:sig ) — sig no contiene la contraseña
  return Utilities.base64EncodeWebSafe(codigo.toLowerCase() + ':' + rol + ':' + sig);
}

// ── Verificar token: devuelve {codigo,rol} o null ────────────
function hmacVerify_(token) {
  if (!token) return null;
  try {
    var dec   = Utilities.newBlob(Utilities.base64DecodeWebSafe(token)).getDataAsString();
    var parts = dec.split(':');
    if (parts.length < 3) return null;
    var codigo = parts[0];
    var rol    = parts[1];
    var sig    = parts.slice(2).join(':');
    // Recomputar firma y comparar
    var expected = hmacSign_(codigo, rol);
    var expParts = Utilities.newBlob(Utilities.base64DecodeWebSafe(expected)).getDataAsString().split(':');
    if (expParts.slice(2).join(':') !== sig) return null;
    return { codigo: codigo, rol: rol };
  } catch(e) { return null; }
}

// ── Auto-inicialización ──────────────────────────────────────
// Se llama al inicio de doGet y doPost. Usa PropertiesService como
// bandera para ejecutar la inicialización UNA SOLA VEZ y no repetir
// el chequeo de hojas en cada petición (costo cero tras la primera vez).
var _INIT_FLAG = 'hc_initialized_v1';
function autoInit_() {
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty(_INIT_FLAG) === 'true') return; // ya inicializado
  try {
    inicializarHC();                         // crea todas las hojas si no existen
    props.setProperty(_INIT_FLAG, 'true');   // marcar como hecho
    Logger.log('AutoInit completado — hojas creadas/verificadas');
  } catch(ex) {
    Logger.log('AutoInit error: ' + ex.message);
    // No lanzar: dejar que la peticion continúe aunque init falle
  }
}

// ── Login ──────────────────────────────────────────────────
function login_(codigo, password) {
  if (!codigo || !password) return {ok:false, error:'Código y contraseña requeridos'};
  var sheet = ss_().getSheetByName(SHEET_HC_USUARIOS);
  if (!sheet) return {ok:false, error:'Hoja de usuarios no encontrada: "'+SHEET_HC_USUARIOS+'". Ejecuta inicializarHC() primero.'};
  var last = sheet.getLastRow();
  if (last < 2) return {ok:false, error:'No hay usuarios registrados'};
  var rows = sheet.getRange(2, 1, last-1, 6).getValues();
  var cod  = String(codigo).toLowerCase().trim();
  var passInput = String(password).trim();
  var passHash  = sha256_(passInput); // hash del input

  for (var i=0; i<rows.length; i++) {
    if (String(rows[i][0]).toLowerCase().trim() !== cod) continue;
    var stored = String(rows[i][1]).trim();
    // Aceptar hash SHA-256 O texto plano (para compatibilidad durante migración)
    var match = (stored === passHash) || (stored === passInput && stored.length < 64);
    if (!match) break;
    // Si la contraseña estaba en texto plano, migrarla automáticamente
    if (stored !== passHash) {
      sheet.getRange(i+2, 2).setValue(passHash);
      SpreadsheetApp.flush();
    }
    var rol    = String(rows[i][2]).trim().toLowerCase();
    var nombre = String(rows[i][3]).trim();
    var token  = hmacSign_(cod, rol); // token HMAC firmado
    return {ok:true, token:token, rol:rol, nombre:nombre, codigo:cod};
  }
  return {ok:false, error:'Código o contraseña incorrectos'};
}

// ── Auth (compatible con CENTYR) ────────────────────────────
function verifyToken_(token) {
  if (!token) return null;
  try {
    // 1. Verificar firma HMAC — sin tocar el Sheet
    var verified = hmacVerify_(token);
    if (!verified) return null;
    // 2. Consultar Sheet para obtener datos frescos del usuario
    var sheet = ss_().getSheetByName(SHEET_HC_USUARIOS);
    if (!sheet) return null;
    var last = sheet.getLastRow();
    if (last < 2) return null;
    var rows = sheet.getRange(2, 1, last-1, 6).getValues();
    for (var i=0; i<rows.length; i++) {
      if (String(rows[i][0]).toLowerCase().trim() === verified.codigo) {
        return {
          codigo:  String(rows[i][0]).trim(),
          rol:     String(rows[i][2]).trim().toLowerCase(),
          nombre:  String(rows[i][3]).trim(),
          ctmp:    String(rows[i][4]||'').trim(),
          firma:   String(rows[i][5]||'').trim()
        };
      }
    }
    return null;
  } catch(e) { return null; }
}

function auth_(p, roles) {
  var u = verifyToken_(p.token || '');
  if (!u) throw new Error('No autorizado');
  if (roles && roles.indexOf(u.rol) === -1) throw new Error('Sin permisos');
  return u;
}

function authB_(tok, roles) {
  var u = verifyToken_(tok);
  if (!u) throw new Error('No autorizado');
  if (roles && roles.indexOf(u.rol) === -1) throw new Error('Sin permisos');
  return u;
}

// ── doGet ────────────────────────────────────────────────────
function doGet(e) {
  autoInit_(); // garantiza que las hojas existan en el primer arranque
  var p = e.parameter || {};
  var a = p.action || '';
  try {
    if (a === 'ping') return ok_({msg:'CENTYR-HC activo'});

    if (a === 'login') return json_(login_(p.codigo, p.password));

    // ── Marca (público — no requiere token para que cargue antes del login) ──
    if (a === 'getBrand') return json_(getBrand_());

    if (a === 'listarCategorias')  return json_(listarCategorias_(p));
    if (a === 'listarPacientes')   return json_(listarPacientes_(p));
    if (a === 'obtenerPaciente')   return json_(obtenerPaciente_(p));
    if (a === 'listarEvaluaciones')return json_(listarEvaluaciones_(p));
    if (a === 'obtenerEvaluacion') return json_(obtenerEvaluacion_(p));
    if (a === 'listarSesiones')    return json_(listarSesiones_(p));
    if (a === 'pendientesRevision')return json_(pendientesRevision_(p));
    if (a === 'listarUsuarios')     return json_(listarUsuarios_(p));
    if (a === 'getConsentimiento')  return json_(getConsentimiento_(p));

    return err_('Acción no reconocida: '+a);
  } catch(ex) {
    return err_(ex.message);
  }
}

// ── doPost ───────────────────────────────────────────────────

  var b = {};
  try { b = JSON.parse(e.postData.contents); } catch(ex) { return err_('JSON inválido'); }
  var a   = b.action || '';
  var tok = b.token  || '';
  try {
    if (a === 'guardarPaciente')    return json_(guardarPaciente_(b, tok));
    if (a === 'eliminarPaciente')   return json_(eliminarPaciente_(b, tok));
    if (a === 'guardarEvaluacion')  return json_(guardarEvaluacion_(b, tok));
    if (a === 'cerrarPorTimer')     return json_(cerrarPorTimer_(b, tok));
    if (a === 'revisarEvaluacion')  return json_(revisarEvaluacion_(b, tok));
    if (a === 'guardarSesion')      return json_(guardarSesion_(b, tok));
    if (a === 'revisarSesion')      return json_(revisarSesion_(b, tok));
    if (a === 'guardarCategoria')   return json_(guardarCategoria_(b, tok));
    if (a === 'reasignarPaciente')      return json_(reasignarPaciente_(b, tok));
    if (a === 'archivarPaciente')       return json_(archivarPaciente_(b, tok));
    if (a === 'guardarFirmaDocente')   return json_(guardarFirmaDocente_(b, tok));
    if (a === 'guardarCtmpDocente')    return json_(guardarCtmpDocente_(b, tok));
    if (a === 'getFirma')              return json_(getFirma_(b));
    // ── Marca ────────────────────────────────────────────────
    if (a === 'uploadLogo')              return json_(uploadLogo_(b, tok));
    if (a === 'updateEmpresa')           return json_(updateEmpresa_(b, tok));
    if (a === 'subirConsentimiento')     return json_(subirConsentimiento_(b, tok));

}

function listarCategorias_(p) {
  try { auth_(p, ['admin','docente','estudiante']); } catch(e) { return {ok:false,error:e.message}; }
  var h    = hojaSegura_(SHEET_HC_CATEGORIAS);
  var last = h.getLastRow();
  if (last < 2) return {ok:true, categorias:[]};
  var rows = h.getRange(2,1,last-1,6).getValues();
  var cats = rows.filter(function(r){ return r[0]; }).map(function(r){
    return {id:String(r[0]),nombre:String(r[1]),descripcion:String(r[2]),
            color:String(r[3]),activa:r[4]===true||r[4]==='true'||r[4]==='TRUE'};
  }).filter(function(c){ return c.activa; });
  return {ok:true, categorias:cats};
}

function guardarCategoria_(b, tok) {
  authB_(tok, ['admin']);
  var h = hojaSegura_(SHEET_HC_CATEGORIAS);
  if (b.id) {
    var rows = h.getDataRange().getValues();
    for (var i=1; i<rows.length; i++) {
      if (String(rows[i][0])===String(b.id)) {
        h.getRange(i+1,1,1,6).setValues([[b.id,b.nombre||rows[i][1],
          b.descripcion||rows[i][2],b.color||rows[i][3],
          b.activa!==undefined?b.activa:rows[i][4],rows[i][5]]]);
        return {ok:true};
      }
    }
  }
  var newId = b.nombre.toLowerCase().replace(/\s+/g,'-').replace(/[^a-z0-9-]/g,'');
  h.appendRow([newId,b.nombre,b.descripcion||'',b.color||'#64748b',true,hoy_()]);
  return {ok:true,id:newId};
}

function getHdrIdx_(h) {
  var hdr = h.getRange(1,1,1,h.getLastColumn()).getValues()[0];
  var idx = {};
  hdr.forEach(function(n,i){ idx[n]=i; });
  return idx;
}

function listarPacientes_(p) {
  var u;
  try { u = auth_(p, ['admin','docente','estudiante']); }
  catch(e) { return {ok:false,error:e.message}; }
  // FIX: eliminada la segunda declaración duplicada de u que sobreescribía con verifyToken_
  var h    = hojaSegura_(SHEET_HC_PACIENTES);
  var last = h.getLastRow();
  if (last < 2) return {ok:true, pacientes:[]};
  var rows = h.getRange(2,1,last-1,20).getValues();
  var idx  = getHdrIdx_(h);

  // Contar evaluaciones pendientes por paciente
  var hE = hojaSegura_(SHEET_HC_EVALUACIONES);
  var idxE = hE.getLastRow()>1 ? getHdrIdx_(hE) : {};
  var evalCols = hE.getLastRow()>1 ? Math.min(hE.getLastColumn(), 40) : 6;
  var evalRows = hE.getLastRow()>1 ? hE.getRange(2,1,hE.getLastRow()-1,evalCols).getValues() : [];
  var evalMap = {};
  var dxFisioMap = {};
  evalRows.forEach(function(r){
    var pid  = String(r[idxE['pacienteId']!==undefined ? idxE['pacienteId'] : 1]||'');
    if (!pid) return;
    if (!evalMap[pid]) evalMap[pid] = {total:0,pendiente:0,aprobada:0};
    evalMap[pid].total++;
    var est  = String(r[idxE['estado']!==undefined ? idxE['estado'] : 4]||'');
    if (est==='pendiente') evalMap[pid].pendiente++;
    if (est==='aprobada')  evalMap[pid].aprobada++;
    var dxF  = idxE['dxFisio']!==undefined ? String(r[idxE['dxFisio']]||'') : '';
    if (est==='aprobada' && dxF && !dxFisioMap[pid]) dxFisioMap[pid] = dxF;
  });

  var pacs = [];
  rows.forEach(function(r){
    if (!r[idx['id']]) return;
    var pid = String(r[idx['id']]);
    var estado = String(r[idx['estado']]||'activo');
    // No mostrar pacientes archivados en la lista normal
    if (estado === 'archivado') return;
    // Filtro por rol
    if (u.rol === 'estudiante' && String(r[idx['creadoPor']])!==u.codigo &&
        String(r[idx['asignadoA']])!==u.codigo) return;
    pacs.push({
      id:                String(r[idx['id']]),
      nombre:            String(r[idx['nombre']]||''),
      dni:               String(r[idx['dni']]||''),
      fechaNac:          String(r[idx['fechaNac']]||''),
      sexo:              String(r[idx['sexo']]||''),
      telefono:          String(r[idx['telefono']]||''),
      ocupacion:         String(r[idx['ocupacion']]||''),
      categoriaId:       String(r[idx['categoriaId']]||''),
      dxMedico:          String(r[idx['dxMedico']]||''),
      dxFisio:           dxFisioMap[pid]||'',
      estado:            estado,
      creadoPor:         String(r[idx['creadoPor']]||''),
      asignadoA:         String(r[idx['asignadoA']]||''),
      fechaInicio:       String(r[idx['fechaInicio']]||''),
      fechaCreacion:     String(r[idx['fechaCreacion']]||''),
      evalTotal:         (evalMap[pid]||{}).total||0,
      evalPendiente:     (evalMap[pid]||{}).pendiente||0,
      evalAprobada:      (evalMap[pid]||{}).aprobada||0,
    });
  });
  pacs.sort(function(a,b){ return a.nombre.localeCompare(b.nombre); });
  return {ok:true, pacientes:pacs};
}

function obtenerPaciente_(p) {
  try { auth_(p, ['admin','docente','estudiante']); } catch(e) { return {ok:false,error:e.message}; }
  if (!p.id) return {ok:false,error:'id requerido'};
  var h = hojaSegura_(SHEET_HC_PACIENTES);
  var rows = h.getDataRange().getValues();
  var idx  = getHdrIdx_(h);
  for (var i=1; i<rows.length; i++) {
    if (String(rows[i][idx['id']])===String(p.id)) {
      var pac = {};
      Object.keys(idx).forEach(function(k){ pac[k]=String(rows[i][idx[k]]||''); });
      return {ok:true, paciente:pac};
    }
  }
  return {ok:false, error:'Paciente no encontrado'};
}

function guardarPaciente_(b, tok) {
  var u = authB_(tok, ['admin','docente','estudiante']);
  var h   = hojaSegura_(SHEET_HC_PACIENTES);
  var idx = getHdrIdx_(h);
  var cols = ['id','nombre','dni','fechaNac','sexo','telefono','ocupacion',
    'responsable','telResponsable','categoriaId','dxMedico','medico',
    'fechaInicio','estado','motivo','antecedentes',
    'creadoPor','asignadoA','fechaCreacion','ultimaActualizacion'];

  if (b.id) {
    var rows = h.getDataRange().getValues();
    for (var i=1; i<rows.length; i++) {
      if (String(rows[i][idx['id']])===String(b.id)) {
        var row = h.getRange(i+1,1,1,cols.length);
        var vals = row.getValues()[0];
        cols.forEach(function(c,ci){
          if (c==='id'||c==='creadoPor'||c==='fechaCreacion') return;
          if (c==='ultimaActualizacion') { vals[ci]=now_(); return; }

        });

      }
    }

function listarEvaluaciones_(p) {
  var u;
  try { u = auth_(p, ['admin','docente','estudiante']); } catch(e) { return {ok:false,error:e.message}; }
  // FIX: eliminada la segunda declaración duplicada de u que sobreescribía con verifyToken_
  var h    = hojaSegura_(SHEET_HC_EVALUACIONES);
  var last = h.getLastRow();
  if (last < 2) return {ok:true, evaluaciones:[]};
  var rows = h.getRange(2,1,last-1,h.getLastColumn()).getValues();
  var idx  = getHdrIdx_(h);

  var evals = [];
  for (var i=0; i<rows.length; i++) {
    var r = rows[i];
    if (!r[idx['id']]) continue;
    if (p.pacienteId && String(r[idx['pacienteId']])!==String(p.pacienteId)) continue;
    var estado = String(r[idx['estado']]||'pendiente');
    var timerIni = idx['timerInicio']!==undefined ? String(r[idx['timerInicio']]||'') : '';
    if (estado==='borrador' && timerIni && (new Date().getTime() - parseInt(timerIni)) > 30*60*1000) {
      // Auto-cerrar en Sheets
      h.getRange(i+2, idx['estado']+1).setValue('vencida');
      SpreadsheetApp.flush();
      estado = 'vencida';
    }
    evals.push({
      id:               String(r[idx['id']]),
      pacienteId:       String(r[idx['pacienteId']]||''),
      categoriaId:      String(r[idx['categoriaId']]||''),
      tipo:             String(r[idx['tipo']]||'inicial'),
      estado:           estado,
      comentarioDocente:String(r[idx['comentarioDocente']]||''),
      revisadoPor:      String(r[idx['revisadoPor']]||''),
      fechaRevision:    String(r[idx['fechaRevision']]||''),
      autor:            String(r[idx['autor']]||''),
      autorCodigo:      String(r[idx['autorCodigo']]||r[idx['autor']]||''),
      docente:          String(r[idx['docente']]||''),
      docenteCodigo:    idx['docenteCodigo']!==undefined ? String(r[idx['docenteCodigo']]||'') : '',
      fecha:            (r[idx['fecha']] instanceof Date)
                          ? Utilities.formatDate(r[idx['fecha']], 'America/Lima', 'yyyy-MM-dd')
                          : String(r[idx['fecha']]||''),
      dxFisio:          String(r[idx['dxFisio']]||''),
      aptaPatron:       String(r[idx['aptaPatron']]||''),
      timerInicio:      timerIni,
    });
  }
  // Ordenar por fecha desc
  evals.sort(function(a,b){ return (b.fecha||'').localeCompare(a.fecha||''); });
  return {ok:true, evaluaciones:evals};
}

function obtenerEvaluacion_(p) {
  try { auth_(p, ['admin','docente','estudiante']); } catch(e) { return {ok:false,error:e.message}; }
  if (!p.id) return {ok:false,error:'id requerido'};
  var h    = hojaSegura_(SHEET_HC_EVALUACIONES);
  var rows = h.getDataRange().getValues();
  var idx  = getHdrIdx_(h);
  for (var i=1; i<rows.length; i++) {
    if (String(rows[i][idx['id']])===String(p.id)) {
      var ev = {};
      Object.keys(idx).forEach(function(k){
        var val = rows[i][idx[k]];
        // Formatear fechas correctamente (evitar "Wed May 13 2026 00:00:00 GMT-0500")
        if (val instanceof Date) {
          ev[k] = Utilities.formatDate(val, 'America/Lima', 'yyyy-MM-dd');
        } else {
          ev[k] = String(val||'');
        }
      });
      // Parsear JSON de rom, fuerza, datosEspecificos
      ['rom','fuerza','datosBase','datosEspecificos'].forEach(function(f){
        try{ if(ev[f]) ev[f]=JSON.parse(ev[f]); } catch(e){ ev[f]={}; }
      });
      // Si widgetPostural viene como string JSON dentro de datosEspecificos, parsearlo
      if (ev.datosEspecificos && typeof ev.datosEspecificos.widgetPostural === 'string' && ev.datosEspecificos.widgetPostural) {
        try { ev.datosEspecificos.widgetPostural = JSON.parse(ev.datosEspecificos.widgetPostural); } catch(e) {}
      }
      if (ev.evaValor!=='') ev.evaValor=Number(ev.evaValor);
      return {ok:true, evaluacion:ev};
    }
  }
  return {ok:false, error:'Evaluación no encontrada'};
}

function guardarEvaluacion_(b, tok) {
  var u   = authB_(tok, ['admin','docente','estudiante']);
  var h   = hojaSegura_(SHEET_HC_EVALUACIONES);

  // FIX: ensureColEval re-lee headers desde Sheets en cada llamada para evitar
  // posiciones incorrectas por copia en memoria desactualizada.
  function ensureColEval(name) {
    var currentHdrs = h.getRange(1, 1, 1, h.getLastColumn()).getValues()[0]
                       .map(function(x){ return String(x).trim(); });
    if (currentHdrs.indexOf(name) === -1) {
      h.getRange(1, h.getLastColumn() + 1).setValue(name);
      SpreadsheetApp.flush();
    }
  }
  ensureColEval('docente');
  ensureColEval('docenteCodigo');
  ensureColEval('autorCodigo');
  ensureColEval('timerInicio');

  var idx  = getHdrIdx_(h);
  var cols = h.getRange(1,1,1,h.getLastColumn()).getValues()[0];

  // Estado inicial: borrador (alumno) o aprobada (docente/admin)
  var estadoInicial = (u.rol==='estudiante') ? 'borrador' : 'aprobada';

  if (b.id) {
    var rows = h.getDataRange().getValues();
    for (var i=1; i<rows.length; i++) {
      if (String(rows[i][idx['id']])===String(b.id)) {
        var row  = h.getRange(i+1,1,1,cols.length);
        var vals = row.getValues()[0];
        var estadoAnterior = String(vals[idx['estado']]||'');
        cols.forEach(function(c,ci){
          if (c==='id'||c==='autor'||c==='fecha'||c==='timerInicio') return;
          // Respetar el estado que manda el frontend; solo resetear si no viene estado
          if (c==='estado'){
            if (b.estado!==undefined) vals[ci]=b.estado;
            // no resetear automáticamente — el frontend controla el estado
            return;
          }
          if (c==='comentarioDocente'||c==='revisadoPor'||c==='fechaRevision') return;
          if (b[c]!==undefined) vals[ci]= typeof b[c]==='object' ? JSON.stringify(b[c]) : b[c];
        });
        // Reiniciar timer al volver a borrador desde rechazada/vencida
        if (b.estado === 'borrador' && estadoAnterior !== 'borrador') {
          var timerColIdx = cols.indexOf('timerInicio');
          if (timerColIdx !== -1) vals[timerColIdx] = String(new Date().getTime());
        }
        row.setValues([vals]);
        return {ok:true,id:b.id};
      }
    }
  }
  var newId = uid_();
  var fila  = cols.map(function(c){
    if (c==='id')          return newId;
    if (c==='autor')       return u.nombre||u.codigo;
    if (c==='autorCodigo') return u.codigo;
    if (c==='fecha')       return b.fecha||hoy_();
    if (c==='estado')      return b.estado||estadoInicial;
    if (c==='timerInicio') return (b.estado==='borrador'||estadoInicial==='borrador') ? String(new Date().getTime()) : '';
    if (c==='comentarioDocente'||c==='revisadoPor'||c==='fechaRevision') return '';
    if (b[c]!==undefined) return typeof b[c]==='object' ? JSON.stringify(b[c]) : b[c];
    return '';
  });
  h.appendRow(fila);
  return {ok:true,id:newId};
}

// Devuelve el contenido del archivo como base64
// El HC lo llama via POST con {action:'getFirma', fileId:'XXX', token:'...'}
function getFirma_(p) {
  auth_(p, ['admin','docente','estudiante']);
  var fileId = String(p.fileId||'').trim();
  if (!fileId) return {ok:false, error:'fileId requerido'};
  try {
    var file     = DriveApp.getFileById(fileId);
    var blob     = file.getBlob();
    var b64      = Utilities.base64Encode(blob.getBytes());
    var mimeType = blob.getContentType()||'image/png';
    return {ok:true, data:b64, mimeType:mimeType, name:file.getName()};
  } catch(ex) {
    return {ok:false, error:'Archivo no encontrado o sin acceso: '+ex.message};
  }
}

  var idx = {};
  hdrs.forEach(function(h,i){ if(h) idx[h] = i; });
  var usuarios = rows
    .filter(function(r){ return String(r[0]).trim() !== ''; })
    .map(function(r){
      return {
        codigo:          String(r[idx['codigo']   !== undefined ? idx['codigo']   : 0]||'').trim(),
        rol:             String(r[idx['rol']       !== undefined ? idx['rol']       : 2]||'').trim().toLowerCase(),
        nombre:          String(r[idx['nombre']    !== undefined ? idx['nombre']    : 3]||'').trim(),
        ctmp:            String(r[idx['ctmp']      !== undefined ? idx['ctmp']      : 4]||'').trim(),
        firma:           String(r[idx['firma']     !== undefined ? idx['firma']     : 5]||'').trim(),
        gradoAcademico:  idx['gradoAcademico'] !== undefined ? String(r[idx['gradoAcademico']]||'').trim() : '',
        cargo:           idx['cargo']           !== undefined ? String(r[idx['cargo']]||'').trim()          : '',
      };
    });
  return {ok:true, usuarios:usuarios};
}

function guardarFirmaDocente_(b, tok) {
  authB_(tok, ['admin']);
  var codigo = String(b.codigo||'').trim();
  var firma  = String(b.firma||'').trim(); // base64 data:image/...
  if (!codigo) return {ok:false, error:'Codigo requerido'};

  var sheet = ss_().getSheetByName(SHEET_HC_USUARIOS);
  if (!sheet) return {ok:false, error:'Hoja de usuarios no encontrada'};
  var last = sheet.getLastRow();
  if (last < 2) return {ok:false, error:'Sin usuarios'};

  // ✅ FIX: Asegurar que existan las columnas básicas (1-4) antes de escribir en 5-6
  // Esto previene conflictos si gashc.txt intenta escribir antes de que gasindex.txt haya creado el usuario
  if (sheet.getLastColumn() < 4) {
    var hdrs = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    if (!hdrs[0] || String(hdrs[0]).trim() === '') sheet.getRange(1, 1).setValue('codigo');
    if (!hdrs[1] || String(hdrs[1]).trim() === '') sheet.getRange(1, 2).setValue('password');
    if (!hdrs[2] || String(hdrs[2]).trim() === '') sheet.getRange(1, 3).setValue('rol');
    if (!hdrs[3] || String(hdrs[3]).trim() === '') sheet.getRange(1, 4).setValue('nombre');
  }

  // Asegurar columnas ctmp(5) y firma(6)
  if (sheet.getLastColumn() < 6) {
    if (sheet.getLastColumn() < 5) sheet.getRange(1, 5).setValue('ctmp');
    sheet.getRange(1, 6).setValue('firma');
  }

  // Si firma vacía → eliminar archivo Drive y limpiar celda
  if (!firma) {
    var col1e = sheet.getRange(2, 1, last-1, 2).getValues();
    for (var ie=0; ie<col1e.length; ie++) {
      if (String(col1e[ie][0]).trim().toLowerCase() === codigo.toLowerCase()) {
        var oldFileId = String(col1e[ie][1]||'').trim();
        if (oldFileId && !oldFileId.startsWith('data:')) {
          try { DriveApp.getFileById(oldFileId).setTrashed(true); } catch(ex) {}
        }
        sheet.getRange(ie+2, 6).setValue('');
        SpreadsheetApp.flush();
        return {ok:true, accion:'firma eliminada'};
      }
    }
    return {ok:false, error:'Usuario no encontrado: '+codigo};
  }

  // Obtener o crear carpeta firmas/
  var folder = getFirmasFolder_();

  // Extraer mimeType y datos del base64
  var mimeType = 'image/png';
  var b64data  = firma;
  var m = firma.match(/^data:([^;]+);base64,(.+)$/);
  if (m) { mimeType = m[1]; b64data = m[2]; }

  // Determinar extensión
  var ext = mimeType === 'image/jpeg' ? '.jpg' : mimeType === 'image/gif' ? '.gif' : '.png';
  var ctmpRow = '';

  // Buscar el CTMP del usuario para el nombre del archivo
  var col1 = sheet.getRange(2, 1, last-1, 5).getValues();
  var rowIdx = -1;
  for (var i=0; i<col1.length; i++) {
    if (String(col1[i][0]).trim().toLowerCase() === codigo.toLowerCase()) {
      rowIdx = i;
      ctmpRow = String(col1[i][4]||'').trim();
      break;
    }
  }
  if (rowIdx === -1) return {ok:false, error:'Usuario no encontrado: '+codigo};

  // Nombre del archivo: CTMP_XXXXXX_codigo.ext o codigo.ext
  var fileName = (ctmpRow ? 'CTMP_'+ctmpRow+'_' : '') + codigo + ext;

  // Eliminar archivo anterior si existe
  var prevFileId = String(sheet.getRange(rowIdx+2, 6).getValue()||'').trim();
  if (prevFileId && !prevFileId.startsWith('data:')) {
    try { DriveApp.getFileById(prevFileId).setTrashed(true); } catch(ex) {}
  }

  // Crear nuevo archivo en Drive
  var blob     = Utilities.newBlob(Utilities.base64Decode(b64data), mimeType, fileName);
  var newFile  = folder.createFile(blob);
  var fileId   = newFile.getId();

  // Guardar solo el fileId en Sheets
  sheet.getRange(rowIdx+2, 6).setValue(fileId);
  SpreadsheetApp.flush();

  return {ok:true, accion:'firma guardada', fileId:fileId, fileName:fileName};
}

// Helper: obtener/crear carpeta firmas/ en Drive

function guardarCtmpDocente_(b, tok) {
  authB_(tok, ['admin']);
  var codigo = String(b.codigo||'').trim();
  var ctmp   = String(b.ctmp||'').trim();
  var grado  = String(b.gradoAcademico||'').trim();
  var cargo  = String(b.cargo||'').trim();
  if (!codigo) return {ok:false, error:'Codigo requerido'};

  var sheet = ss_().getSheetByName(SHEET_HC_USUARIOS);
  if (!sheet) return {ok:false, error:'Hoja de usuarios no encontrada'};
  var last = sheet.getLastRow();
  if (last < 2) return {ok:false, error:'Sin usuarios'};

  // FIX: ensureCol re-lee los headers desde Sheets en cada llamada para evitar
  // calcular posiciones incorrectas basándose en una copia en memoria desactualizada.
  function ensureCol(name) {
    var currentHdrs = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]
                           .map(function(h){ return String(h).trim(); });
    var idx = currentHdrs.indexOf(name);
    if (idx === -1) {
      var newCol = sheet.getLastColumn() + 1;
      sheet.getRange(1, newCol).setValue(name);
      SpreadsheetApp.flush(); // forzar escritura antes de continuar
      return newCol;
    }
    return idx + 1; // 1-indexed
  }

  ensureCol('ctmp');
  ensureCol('firma');
  var colGrado = ensureCol('gradoAcademico');
  var colCargo = ensureCol('cargo');

  // Re-leer headers finales ya con todas las columnas aseguradas
  var finalHdrs = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]
                       .map(function(h){ return String(h).trim(); });
  var colCtmp = finalHdrs.indexOf('ctmp') + 1;

  // Buscar la fila del usuario
  var rows = sheet.getRange(2, 1, last-1, 1).getValues();
  for (var i=0; i<rows.length; i++) {
    if (String(rows[i][0]).trim().toLowerCase() === codigo.toLowerCase()) {
      var row = i + 2;
      if (ctmp  !== '') sheet.getRange(row, colCtmp).setValue(ctmp);
      if (grado !== '') sheet.getRange(row, colGrado).setValue(grado);
      if (cargo !== '') sheet.getRange(row, colCargo).setValue(cargo);
      // Renombrar archivo Drive si cambió el CTMP
      if (ctmp) {
        var firmaIdx = finalHdrs.indexOf('firma');
        if (firmaIdx !== -1) {
          var fileId = String(sheet.getRange(row, firmaIdx+1).getValue()||'').trim();
          if (fileId && !fileId.startsWith('data:')) {
            try {
              var file = DriveApp.getFileById(fileId);
              var oldName = file.getName();
              var ext = oldName.substring(oldName.lastIndexOf('.')) || '.png';
              file.setName('CTMP_'+ctmp+'_'+codigo+ext);
            } catch(ex) {}
          }
        }
      }
      SpreadsheetApp.flush();
      return {ok:true, ctmp:ctmp, gradoAcademico:grado, cargo:cargo};
    }
  }
  return {ok:false, error:'Usuario no encontrado: '+codigo};
}

// ── Redimensionar PNG preservando transparencia ───────────────
// Usa Drive Thumbnail API: sube temp → obtiene thumbnail → elimina temp
// Si falla cualquier paso, devuelve el base64 original sin modificar (fallback seguro)
function resizePngBase64_(base64Input, maxPx) {
  maxPx = maxPx || 200;

  // Validar formato base64
  var matches = base64Input.match(/^data:([^;]+);base64,(.+)$/);
  if (!matches) throw new Error('Formato base64 inválido — debe incluir el prefijo data:image/...;base64,');

  var mimeType = matches[1]; // ej: image/png, image/jpeg
  var rawB64   = matches[2];

  // Decodificar y crear blob
  var bytes = Utilities.base64Decode(rawB64);
  var blob  = Utilities.newBlob(bytes, mimeType, 'logo_tmp');

  try {
    // 1. Subir temporalmente a Drive
    var tempFile = DriveApp.createFile(blob);
    var fileId   = tempFile.getId();

    // 2. Obtener thumbnail redimensionado (Drive Thumbnail API — máx 800)
    var size     = Math.min(Math.max(maxPx, 32), 800);
    var thumbUrl = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=s' + size;
    var response = UrlFetchApp.fetch(thumbUrl, {muteHttpExceptions: true});

    var resizedB64;
    if (response.getResponseCode() === 200) {
      var resizedBytes = response.getBlob().getBytes();
      resizedB64 = 'data:image/png;base64,' + Utilities.base64Encode(resizedBytes);
    } else {
      // Thumbnail no disponible — usar original
      Logger.log('Thumbnail no disponible (HTTP ' + response.getResponseCode() + '), usando original');
      resizedB64 = base64Input;
    }

    // 3. Eliminar archivo temporal de Drive
    try { DriveApp.getFileById(fileId).setTrashed(true); } catch(ex) {}

    return resizedB64;

  } catch(ex) {
    // Fallback seguro: devolver base64 original sin redimensionar
    Logger.log('resizePngBase64_ falló (' + ex.message + '), usando imagen original');
    return base64Input;
  }
}

/**
 * migrarContrasenas — convierte TODAS las contraseñas en texto plano
 * a SHA-256. Ejecutar UNA SOLA VEZ después de desplegar.
 * Es seguro volver a ejecutar: si ya son hash (64 chars hex) las omite.
 */
function migrarContrasenas() {
  var sheet = ss_().getSheetByName(SHEET_HC_USUARIOS);
  if (!sheet) { Logger.log('ERROR: Hoja _usuarios no encontrada'); return; }
  var last = sheet.getLastRow();
  if (last < 2) { Logger.log('Sin usuarios'); return; }
  var rows = sheet.getRange(2, 1, last-1, 2).getValues();
  var migradas = 0, omitidas = 0;
  for (var i=0; i<rows.length; i++) {
    var cod    = String(rows[i][0]).trim();
    var stored = String(rows[i][1]).trim();
    if (!cod) continue;
    // Si ya tiene 64 chars hexadecimales → ya es SHA-256 → omitir
    if (/^[0-9a-f]{64}$/.test(stored)) { omitidas++; continue; }
    // Texto plano → hashear
    sheet.getRange(i+2, 2).setValue(sha256_(stored));
    migradas++;
  }
  SpreadsheetApp.flush();
  Logger.log('Migración completa. Migradas: ' + migradas + ' | Ya hasheadas: ' + omitidas);
  Logger.log('IMPORTANTE: A partir de ahora los usuarios deben usar su contraseña ORIGINAL (el hash se aplica automáticamente en el login).');
}
