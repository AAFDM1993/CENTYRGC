// AUTH
// ════════════════════════════════════════════════════════
function login(codigo, password) {
  if (!codigo || !password) return { error: 'Codigo y contrasena requeridos' };
  var user = findUser(codigo);
  if (!user) return { error: 'Usuario no encontrado' };
  var passwordIngresadaHash = generarHashSHA256(password);
  if (String(user.password) !== passwordIngresadaHash) return { error: 'Contrasena incorrecta' };
  var token = Utilities.base64Encode(user.codigo + ':' + user.rol + ':' + passwordIngresadaHash);
  return { ok: true, token: token, rol: user.rol, nombre: user.nombre, codigo: user.codigo };
}

function findUser(codigo) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_USUARIOS);
  if (!sheet) return null;
  var last = sheet.getLastRow();
  if (last < 2) return null;
  var rows   = sheet.getRange(2, 1, last - 1, 4).getValues();
  var needle = String(codigo).trim().toLowerCase();
  for (var i = 0; i < rows.length; i++) {
    var rowCod = String(rows[i][0]).trim();
    if (rowCod.toLowerCase() === needle) {
      return {
        codigo:   rowCod,
        password: String(rows[i][1]),
        rol:      String(rows[i][2]).trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g,''),
        nombre:   String(rows[i][3])
      };
    }
  }
  return null;
}

function verifyToken(token) {
  if (!token) return null;
  try {
    var dec   = Utilities.newBlob(Utilities.base64Decode(token)).getDataAsString();
    var parts = dec.split(':');
    if (parts.length < 3) return null;
    var codigo       = parts[0];
    var rol          = parts[1];
    var passwordHash = parts[2];
    var user = findUser(codigo);
    if (!user) return null;
    if (String(user.password) !== String(passwordHash)) return null;
    return user;
  } catch(e) { return null; }
}

function requireAuth(p, roles, fn) {
  var user = verifyToken(p.token || '');
  if (!user) return { error: 'No autorizado', code: 401 };
  if (roles.indexOf(user.rol) === -1) return { error: 'Sin permisos', code: 403 };
  return fn(user);
}

function requireAuthBody(token, roles, fn) {
  var user = verifyToken(token);
  if (!user) return { error: 'No autorizado', code: 401 };
  if (roles.indexOf(user.rol) === -1) return { error: 'Sin permisos', code: 403 };
  return fn(user);
}

function listarHojas() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var hojas = ss.getSheets()
    .filter(function(s){ return s.getName().indexOf('_') !== 0; })
    .map(function(s){ return { nombre: s.getName(), alumnos: contarAlumnos(s) }; });
  return { ok: true, hojas: hojas };
}

function contarAlumnos(sheet) {
  var last = sheet.getLastRow();
  if (last < 1) return 0;
  var vals = sheet.getRange(1, 1, last, 1).getValues();
  var n = 0;
  for (var i = 0; i < vals.length; i++) {
    if (String(vals[i][0]).indexOf('ALUMNO:') === 0) n++;
  }
  return n;
}

// LEER HOJA
// ════════════════════════════════════════════════════════
function leerHoja(nombre) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(nombre);
  if (!sheet) return { error: 'Hoja no encontrada: ' + nombre };
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow < 3) return { ok: true, alumnos: [], config: {} };
  var config = {};
  try { config = JSON.parse(sheet.getRange(1, 1).getValue()); } catch (e) {}
  var rows = sheet.getRange(3, 1, lastRow - 2, lastCol).getValues();
  var alumnos = [], curAl = null, curCu = null, curSgr = null;
  for (var ri = 0; ri < rows.length; ri++) {
    var c0 = String(rows[ri][0]).trim();
    if (c0.indexOf('ALUMNO:') === 0) {
      if (curAl) alumnos.push(curAl);
      curAl  = { nombre: c0.slice(7).trim(), codigo: String(rows[ri][1]).trim(), cursos: [] };
      curCu  = null; curSgr = null;
    } else if (c0.indexOf('CURSO:') === 0 && curAl) {
      curCu  = { nombre: c0.slice(6).trim(), subgrupos: [] };
      curAl.cursos.push(curCu);
      curSgr = null;
    } else if (c0.indexOf('SGR:') === 0 && curCu) {
      var sgrParts  = c0.slice(4).split('|');
      var sBase     = sgrParts.length >= 2 ? (Number(sgrParts[1]) || 0) : 0;
      var sExtra    = sgrParts.length >= 3 ? (Number(sgrParts[2]) || 0) : 0;
      var sPctExtra = sgrParts.length >= 4 ? (Number(sgrParts[3]) || 0) : (Number(config.pctExtra) || 0);
      curSgr = { nombre: sgrParts[0].trim(), notasBase: sBase, notasExtra: sExtra, pctExtra: sPctExtra, pacientes: [] };
      curCu.subgrupos.push(curSgr);
    } else if (c0.indexOf('PAC') === 0 && curSgr) {
      var base   = curSgr.notasBase;
      var extra  = curSgr.notasExtra;
      var nomPac = String(rows[ri][1]).trim();
      var notas  = rows[ri].slice(2, 2 + base).map(function(v){ return v === '' ? '' : Number(v); });
      var xNotas = rows[ri].slice(2 + base, 2 + base + extra).map(function(v){ return v === '' ? '' : Number(v); });
      var promVal   = rows[ri][lastCol - 2];
      var notaFinal = rows[ri][lastCol - 1];
      curSgr.pacientes.push({
        label:    c0,
        nombre:   nomPac,
        notas:    notas,
        xNotas:   xNotas,
        promSS:   (promVal   === '' || promVal   === null) ? '' : Number(promVal),
        prom:     (notaFinal === '' || notaFinal === null) ? '' : Number(notaFinal),
        filaReal: ri + 3
      });
    }
  }
  if (curAl) alumnos.push(curAl);
  return { ok: true, alumnos: alumnos, config: config };
}

// MIS NOTAS (estudiante)
// ════════════════════════════════════════════════════════
function misNotas(codigoEstudiante) {
  var ss        = SpreadsheetApp.getActiveSpreadsheet();
  var sheets    = ss.getSheets().filter(function(s){ return s.getName().indexOf('_') !== 0; });
  var resultado = [];
  var codNeedle = String(codigoEstudiante).trim().toLowerCase();
  var docenteMap = {};
  try {
    var logSheet = getOrCreateLogSheet();
    var logLast  = logSheet.getLastRow();
    if (logLast >= 2) {
      var logRows = logSheet.getRange(2, 1, logLast - 1, 11).getValues();
      for (var li = logRows.length - 1; li >= 0; li--) {
        var lr      = logRows[li];
        var docente = String(lr[1]).trim();
        if (!docente) continue;
        var key = [String(lr[2]).trim().toLowerCase(), String(lr[3]).trim().toLowerCase(),
                   String(lr[4]).trim().toLowerCase(), String(lr[5]).trim().toLowerCase(),
                   String(lr[6]).trim().toLowerCase(), String(lr[7]).trim().toLowerCase(),
                   String(lr[8]).trim()].join('||');
        if (!docenteMap[key]) docenteMap[key] = docente;
      }
    }
  } catch(e) {}
  sheets.forEach(function(sheet) {
    var config = {};
    try { config = JSON.parse(sheet.getRange(1,1).getValue()); } catch(e) {}
    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    if (lastRow < 3 || lastCol < 5) return;
    var rows       = sheet.getRange(3, 1, lastRow - 2, lastCol).getValues();
    var encontrado = false;
    var curAl = null, curCu = null, curSgr = null;
    var alumnoNombreLow = '';
    var hojaNombreLow = sheet.getName().trim().toLowerCase();
    for (var ri = 0; ri < rows.length; ri++) {
      var c0 = String(rows[ri][0]).trim();
      if (c0.indexOf('ALUMNO:') === 0) {
        if (encontrado && curAl) resultado.push(curAl);
        encontrado = false; curAl = null; curCu = null; curSgr = null; alumnoNombreLow = '';
        var cod = String(rows[ri][1]).trim().toLowerCase();
        if (cod === codNeedle) {
          encontrado = true;
          alumnoNombreLow = c0.slice(7).trim().toLowerCase();
          curAl = { hoja: sheet.getName(), asignatura: config.asignatura || sheet.getName(), semestre: config.semestre || '', cursos: [] };
        }
      } else if (encontrado) {
        if (c0.indexOf('CURSO:') === 0) {
          curCu  = { nombre: c0.slice(6).trim(), subgrupos: [] };
          curAl.cursos.push(curCu);
          curSgr = null;
        } else if (c0.indexOf('SGR:') === 0 && curCu) {
          var sgrP = c0.slice(4).split('|');
          curSgr = { nombre: sgrP[0].trim(), notasBase: sgrP.length >= 2 ? (Number(sgrP[1]) || 0) : 0,
                     notasExtra: sgrP.length >= 3 ? (Number(sgrP[2]) || 0) : 0,
                     pctExtra:   sgrP.length >= 4 ? (Number(sgrP[3]) || 0) : (Number(config.pctExtra) || 0), pacientes: [] };
          curCu.subgrupos.push(curSgr);
        } else if (c0.indexOf('PAC') === 0 && curSgr) {
          var base   = curSgr.notasBase;
          var extra  = curSgr.notasExtra;
          var notas  = rows[ri].slice(2, 2+base).map(function(v){ return v===''?'':Number(v); });
          var xNotas = rows[ri].slice(2+base, 2+base+extra).map(function(v){ return v===''?'':Number(v); });
          var curCuLow  = curCu.nombre.trim().toLowerCase();
          var curSgrLow = curSgr.nombre.trim().toLowerCase();
          var c0Low     = c0.toLowerCase();
          var notasDoc = notas.map(function(v, ni) {
            if (v === '') return '';
            var k = [hojaNombreLow, alumnoNombreLow, curCuLow, curSgrLow, c0Low, 'ss', String(ni+1)].join('||');
            return docenteMap[k] || '';
          });
          var xNotasDoc = xNotas.map(function(v, ni) {
            if (v === '') return '';
            var k = [hojaNombreLow, alumnoNombreLow, curCuLow, curSgrLow, c0Low, 'extra', String(ni+1)].join('||');
            return docenteMap[k] || '';
          });
          var promVal   = rows[ri][lastCol - 2];
          var notaFinal = rows[ri][lastCol - 1];
          curSgr.pacientes.push({
            label: c0, nombre: String(rows[ri][1]).trim(),
            notas: notas, xNotas: xNotas, notasDoc: notasDoc, xNotasDoc: xNotasDoc,
            promSS:    (promVal   === '' || promVal   === null) ? '' : Number(promVal),
            prom:      (notaFinal === '' || notaFinal === null) ? '' : Number(notaFinal)
          });
        }
      }
    }
    if (encontrado && curAl) resultado.push(curAl);
  });
  return { ok: true, resultados: resultado };
}

function crearPlanilla(body) {
  var config     = body.config  || {};
  var alumnos    = body.alumnos || [];
  var nombreHoja = body.nombreHoja || '';
  var pctExtra   = Number(config.pctExtra) || 20;
  var ss     = SpreadsheetApp.getActiveSpreadsheet();
  var titulo = (nombreHoja || ((config.seccion || '') + ' ' + (config.asignatura || '').substring(0, 20))).trim() || 'Planilla';
  if (ss.getSheetByName(titulo))
    titulo = titulo + ' (' + Utilities.formatDate(new Date(), 'GMT-5', 'HH:mm') + ')';
  var sheet = ss.insertSheet(titulo);
  var maxCols = 0;
  for (var ai = 0; ai < alumnos.length; ai++) {
    for (var ci = 0; ci < alumnos[ai].cursos.length; ci++) {
      for (var si = 0; si < alumnos[ai].cursos[ci].subgrupos.length; si++) {
        var sg  = alumnos[ai].cursos[ci].subgrupos[si];
        var tot = Number(sg.notasBase) + Number(sg.notasExtra);
        if (tot > maxCols) maxCols = tot;
      }
    }
  }
  if (maxCols < 1) maxCols = 1;
  var colProm      = 2 + maxCols + 1;
  var colNotaFinal = 2 + maxCols + 2;
  var totalCols    = colNotaFinal;
  sheet.getRange(1, 1).setValue(JSON.stringify(config));
  sheet.hideRows(1);
  var hdrs = ['ALUMNO / PACIENTE', 'NOMBRE PACIENTE'];
  for (var h = 1; h <= maxCols; h++) hdrs.push('COL' + h);
  hdrs.push('PROM', 'NOTA FINAL');
  sheet.getRange(2, 1, 1, totalCols).setValues([hdrs]);
  sheet.getRange(2, 1, 1, 2).setBackground('#0f172a').setFontColor('#94a3b8').setFontWeight('bold').setFontSize(10);
  sheet.getRange(2, 3, 1, maxCols).setBackground('#1e3a5f').setFontColor('#93c5fd').setFontWeight('bold').setFontSize(10).setHorizontalAlignment('center');
  sheet.getRange(2, colProm,      1, 1).setBackground('#0c4a6e').setFontColor('#7dd3fc').setFontWeight('bold').setFontSize(10).setHorizontalAlignment('center');
  sheet.getRange(2, colNotaFinal, 1, 1).setBackground('#1d4ed8').setFontColor('#fff').setFontWeight('bold').setFontSize(10).setHorizontalAlignment('center');
  sheet.setRowHeight(2, 26);
  sheet.setFrozenRows(2);
  sheet.setFrozenColumns(2);
  var fila = 3;
  for (var ai = 0; ai < alumnos.length; ai++) {
    var al = alumnos[ai];
    sheet.getRange(fila, 1).setValue('ALUMNO:' + al.nombre);
    sheet.getRange(fila, 2).setValue(al.codigo || '');
    sheet.getRange(fila, 1, 1, totalCols).setBackground('#0f172a').setFontColor('#ffffff').setFontWeight('bold').setFontSize(11);
    sheet.setRowHeight(fila, 28);
    fila++;
    for (var ci = 0; ci < al.cursos.length; ci++) {
      var cu = al.cursos[ci];
      sheet.getRange(fila, 1).setValue('CURSO:' + cu.nombre);
      sheet.getRange(fila, 1, 1, totalCols).setBackground('#1e3a5f').setFontColor('#bfdbfe').setFontWeight('bold').setFontSize(10).setFontStyle('italic');
      sheet.setRowHeight(fila, 24);
      fila++;
      for (var si = 0; si < cu.subgrupos.length; si++) {
        var sgr        = cu.subgrupos[si];
        var notasBase  = Math.max(1, Number(sgr.notasBase)  || 1);
        var notasExtra = Math.max(0, Number(sgr.notasExtra) || 0);
        var totalSS    = notasBase + notasExtra;
        var sgrPct     = Number(sgr.pctExtra) || pctExtra;
        sheet.getRange(fila, 1).setValue('SGR:' + sgr.nombre + '|' + notasBase + '|' + notasExtra + '|' + sgrPct);
        sheet.getRange(fila, 1, 1, totalCols).setBackground('#1d4ed8').setFontColor('#e0f2fe').setFontWeight('bold').setFontSize(10);
        for (var h = 1; h <= totalSS; h++) {
          sheet.getRange(fila, 2 + h).setValue(h === 1 ? 'EV' : ('S' + (h - 1)));
        }
        sheet.getRange(fila, colProm).setValue('PROM');
        sheet.getRange(fila, colNotaFinal).setValue('NOTA FINAL');
        sheet.setRowHeight(fila, 22);
        fila++;
        var numPac     = Math.max(1, Number(sgr.numPacientes) || 1);
        var filaPacIni = fila;
        var filaPacFin = fila + numPac - 1;
        for (var pi = 1; pi <= numPac; pi++) {
          escribirFilaPac_(sheet, fila, pi, false, notasBase, notasExtra, maxCols, colProm, colNotaFinal, filaPacIni, filaPacFin, numPac);
          fila++;
        }
      }
    }
    sheet.getRange(fila, 1, 1, totalCols).setBackground('#1e3a5f');
    sheet.setRowHeight(fila, 5);

// HELPER: escribir fila de paciente
// ════════════════════════════════════════════════════════
function escribirFilaPac_(sheet, rowNum, pi, esExtra, notasBase, notasExtra, maxCols, colProm, colNotaFinal, filaPacIni, filaPacFin, numPac) {
  var totalSS = notasBase + notasExtra;
  sheet.getRange(rowNum, 1).setValue(esExtra ? 'PACX' + pi : 'PAC' + pi);
  sheet.getRange(rowNum, 2).setValue('');
  if (totalSS > 0)
    sheet.getRange(rowNum, 3, 1, totalSS).setValue('').setHorizontalAlignment('center');
  var sobra = maxCols - totalSS;
  if (sobra > 0)
    sheet.getRange(rowNum, 3 + totalSS, 1, sobra).setBackground('#e8edf8');
  var rngSS    = colLetra_(3) + rowNum + ':' + colLetra_(2 + notasBase) + rowNum;
  var rngExtra = notasExtra > 0 ? colLetra_(3 + notasBase) + rowNum + ':' + colLetra_(2 + totalSS) + rowNum : null;
  var fProm;
  if (esExtra) {
    var rngTodo = colLetra_(3) + rowNum + ':' + colLetra_(2 + totalSS) + rowNum;
    fProm = '=IFERROR(ROUND(AVERAGEIF(' + rngTodo + ',"<>"),2),"")';
  } else if (notasExtra > 0) {
    var fPromSS    = 'SUM(' + rngSS + ')/' + notasBase;
    var fPromExtra = 'SUM(' + rngExtra + ')/' + notasExtra;
    fProm = '=IFERROR(ROUND(IF(COUNTIF(' + rngExtra + ',"<>")>0,(' + fPromSS + '+' + fPromExtra + ')/2,' + fPromSS + '),2),"")';
  } else {
    fProm = '=IFERROR(ROUND(SUM(' + rngSS + ')/' + notasBase + ',2),"")';
  }
  sheet.getRange(rowNum, colProm).setFormula(fProm);
  var rngPromAll = colLetra_(colProm) + filaPacIni + ':' + colLetra_(colProm) + filaPacFin;
  var fFinal = '=IFERROR(ROUND(AVERAGEIF(' + rngPromAll + ',"<>"),2),"")';
  sheet.getRange(rowNum, colNotaFinal).setFormula(fFinal);
  var bg = (pi % 2 === 0) ? '#e8edf8' : '#f0f4ff';
  var fg = '#1e3a5f';
  sheet.getRange(rowNum, 1, 1, 2).setBackground(bg).setFontColor(fg).setFontSize(10);
  if (totalSS > 0)
    sheet.getRange(rowNum, 3, 1, totalSS).setBackground(bg).setFontColor(fg).setHorizontalAlignment('center');
  sheet.getRange(rowNum, colProm).setBackground('#0c4a6e').setFontColor('#7dd3fc').setFontWeight('bold').setHorizontalAlignment('center');
  sheet.getRange(rowNum, colNotaFinal).setBackground('#1e3a8a').setFontColor('#fff').setFontWeight('bold').setHorizontalAlignment('center');
  sheet.setRowHeight(rowNum, 22);
}

function guardarNota(body, user) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(body.hoja);
  if (!sheet) return { error: 'Hoja no encontrada' };
  var notasBase  = Number(body.notasBase)  || 1;

  var idx        = Number(body.idx);
  var tipo       = body.tipo;

  if      (tipo === 'ss')    col = 3 + idx;
  else if (tipo === 'extra') col = 3 + notasBase + idx;
  else return { error: 'Tipo desconocido: ' + tipo };
  var valorAnterior = sheet.getRange(fila, col).getValue();
  var val = (body.valor === '' || body.valor === null || body.valor === undefined) ? '' : parseFloat(body.valor);
  sheet.getRange(fila, col).setValue(val);
  SpreadsheetApp.flush();
  try {
    var docente = user ? (user.nombre || user.codigo) : 'desconocido';
    appendLog_(body.hoja, docente, body.alumno, body.curso, body.subgrupo, body.pacLabel, tipo, idx + 1, valorAnterior, val);
  } catch(e) {}
  var lastCol   = sheet.getLastColumn();
  var notaFinal = sheet.getRange(fila, lastCol).getValue();
  var promVal   = sheet.getRange(fila, lastCol - 1).getValue();
  return {
    ok:     true,
    prom:   (notaFinal === '' || notaFinal === null) ? '' : Number(notaFinal),
    promSS: (promVal   === '' || promVal   === null) ? '' : Number(promVal)
  };
}

// ════════════════════════════════════════════════════════
// RESUMEN POR HOJA
// ════════════════════════════════════════════════════════
function resumenHoja(nombre) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(nombre);
  if (!sheet) return { error: 'Hoja no encontrada: ' + nombre };
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow < 3) return { ok: true, hoja: nombre, config: {}, cursos: [] };
  var config = {};
  try { config = JSON.parse(sheet.getRange(1,1).getValue()); } catch(e) {}
  var rows = sheet.getRange(3, 1, lastRow - 2, lastCol).getValues();
  var cursosMap = {}, cursosOrd = [];
  var curCurso = null, curSgr = null, curAlNom = '', curAlCod = '', enAlumno = false;
  for (var ri = 0; ri < rows.length; ri++) {
    var c0 = String(rows[ri][0]).trim();
    if (c0.indexOf('ALUMNO:') === 0) {
      curAlNom = c0.slice(7).trim(); curAlCod = String(rows[ri][1]).trim();
      enAlumno = true; curCurso = null; curSgr = null;
    } else if (enAlumno && c0.indexOf('CURSO:') === 0) {
      curCurso = c0.slice(6).trim(); curSgr = null;
      if (!cursosMap[curCurso]) { cursosMap[curCurso] = { nombre: curCurso, alumnos: {} }; cursosOrd.push(curCurso); }
      if (!cursosMap[curCurso].alumnos[curAlCod])
        cursosMap[curCurso].alumnos[curAlCod] = { nombre: curAlNom, codigo: curAlCod, notasFinal: [] };
    } else if (enAlumno && curCurso && c0.indexOf('SGR:') === 0) {
      var sgrR = c0.slice(4).split('|');
      curSgr   = { notasBase: Number(sgrR[1])||0, notasExtra: Number(sgrR[2])||0, pctExtra: Number(sgrR[3])||0 };
    } else if (enAlumno && curCurso && curSgr && c0.indexOf('PAC') === 0) {
      var nf = rows[ri][lastCol - 1];
      if (nf !== '' && nf !== null && nf !== undefined && !isNaN(Number(nf))) {
        cursosMap[curCurso].alumnos[curAlCod].notasFinal.push(Number(nf));
      }
    }
  }
  var cursosResult = cursosOrd.map(function(cn) {
    var cu = cursosMap[cn];
    var alumnos = Object.keys(cu.alumnos).map(function(cod) {
      var al = cu.alumnos[cod];
      var pf = '';
      if (al.notasFinal.length > 0) {
        var suma = al.notasFinal.reduce(function(a,b){ return a+b; }, 0);
        pf = Math.round(suma / al.notasFinal.length * 100) / 100;
      }
      return { nombre: al.nombre, codigo: al.codigo, promFinal: pf };
    });
    return { nombre: cn, alumnos: alumnos };
  });
  return { ok: true, hoja: nombre, config: config, cursos: cursosResult };
}

function guardarUsuario(b) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_USUARIOS);
  if (!sheet) return { error: 'Hoja de usuarios no encontrada' };
  var codigoNuevo = String(b.codigo).trim();
  var passHash    = generarHashSHA256(String(b.password).trim());
  var rolNuevo    = String(b.rol || 'estudiante').toLowerCase();
  var nombreNuevo = String(b.nombre || '').trim();
  var last   = sheet.getLastRow();
  var needle = codigoNuevo.toLowerCase();
  var filaExistente = -1;
  if (last >= 2) {
    var col1 = sheet.getRange(2, 1, last - 1, 1).getValues();
    for (var i = 0; i < col1.length; i++) {
      if (String(col1[i][0]).trim().toLowerCase() === needle) { filaExistente = i + 2; break; }
    }
  }
  if (filaExistente > 0) {
    // ✅ FIX: Implementar merge en lugar de sobrescritura
    // Leer la fila completa para preservar campos adicionales (ctmp, firma, gradoAcademico, cargo)
    var numCols = sheet.getLastColumn();
    var filaActual = sheet.getRange(filaExistente, 1, 1, numCols).getValues()[0];
    
    // Hacer merge: actualizar solo columnas 1-4, preservar el resto
    var filaActualizada = [];
    for (var c = 0; c < numCols; c++) {
      if (c === 0) filaActualizada[c] = codigoNuevo;           // col 1: codigo
      else if (c === 1) filaActualizada[c] = passHash;         // col 2: password
      else if (c === 2) filaActualizada[c] = rolNuevo;         // col 3: rol
      else if (c === 3) filaActualizada[c] = nombreNuevo;      // col 4: nombre
      else filaActualizada[c] = filaActual[c];                 // cols 5+: preservar
    }
    
    sheet.getRange(filaExistente, 1, 1, numCols).setValues([filaActualizada]);
  } else {
    sheet.appendRow([codigoNuevo, passHash, rolNuevo, nombreNuevo]);
  }
  SpreadsheetApp.flush();
  sincronizarUsuariosADestino_();
  return { ok: true, accion: filaExistente > 0 ? 'actualizado' : 'creado' };
}

// ════════════════════════════════════════════════════════
function buscarFila_(sheet, alumno, curso, subgrupo, pacLabel) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 3) return -1;
  var col1 = sheet.getRange(3, 1, lastRow-2, 1).getValues();
  var enAl = false, enCu = false, enSgr = false;
  for (var ri = 0; ri < col1.length; ri++) {
    var c0 = String(col1[ri][0]).trim();
    if (c0 === 'ALUMNO:' + alumno)             { enAl=true;  enCu=false; enSgr=false; continue; }
    if (enAl && c0.indexOf('ALUMNO:') === 0)   { return -1; }
    if (enAl && c0 === 'CURSO:' + curso)        { enCu=true;  enSgr=false; continue; }
    if (enAl && enCu && c0.indexOf('CURSO:') === 0) { enCu=false; enSgr=false; continue; }
    if (enAl && enCu && c0.indexOf('SGR:') === 0) {
      var _sn = c0.slice(4).split('|')[0].trim().toLowerCase().normalize('NFC');
      if (_sn === String(subgrupo).trim().toLowerCase().normalize('NFC')) { enSgr=true; continue; }
    }
    if (enAl && enCu && enSgr && c0.indexOf('SGR:') === 0) { enSgr=false; continue; }
    if (enAl && enCu && enSgr && c0 === pacLabel)           { return ri + 3; }
  }
  return -1;
}

function colLetra_(n) {
  var s = '';
  while (n > 0) { n--; s = String.fromCharCode(65 + (n%26)) + s; n = Math.floor(n/26); }
  return s;
}

function horaToStr_(v) {
  if (v instanceof Date) return ('0'+v.getHours()).slice(-2)+':'+('0'+v.getMinutes()).slice(-2);
  if (typeof v === 'number' && v > 0 && v < 1) {
    var h = Math.floor(v * 24);
    var m = Math.round((v * 24 - h) * 60);
    return ('0'+h).slice(-2)+':'+('0'+m).slice(-2);
  }
  return String(v).trim();
}

      horaInicio:   horaToStr_(r[1]),
      area:         String(r[2]).trim(),
      camilla:      String(r[3]).trim(),
      paciente:     String(r[4]).trim(),
      docente:      String(r[5]).trim(),
      observacion:  String(r[6]).trim(),
      reservadoPor: String(r[7]).trim(),
      ts:           String(r[8]).trim()
    };
  });
  return { ok: true, reservas: reservas };
}

function crearReserva(body, user) {
  if (!body.fecha || !body.horaInicio || !body.area || !body.camilla)
    return { error: 'Faltan campos requeridos: fecha, horaInicio, area, camilla' };
  var sheet = getAgendaSheet();
  var last  = sheet.getLastRow();
  var quien = user ? (user.nombre || user.codigo) : '';
  var bSheet = getBloqueoSheet();
  var bLast  = bSheet.getLastRow();
  if (bLast >= 2) {
    var bRows = bSheet.getRange(2,1,bLast-1,4).getValues();
    for (var bi = 0; bi < bRows.length; bi++) {
      var bFecha   = bRows[bi][0]==='*'?'*':(bRows[bi][0]?Utilities.formatDate(new Date(bRows[bi][0]),'GMT-5','yyyy-MM-dd'):'');
      var bHora    = String(bRows[bi][1]).trim();
      var bArea    = String(bRows[bi][2]).trim();
      var bCamilla = String(bRows[bi][3]).trim();
      var fechaOk  = bFecha==='*' || bFecha===body.fecha;
      var horaOk   = bHora==='*'  || bHora===body.horaInicio;
      var areaOk   = bArea==='*'  || bArea===body.area;
      var camOk    = bCamilla==='*' || bCamilla===body.camilla;
      if (fechaOk && horaOk && areaOk && camOk)
        return { error: 'Este espacio está bloqueado: ' + (bRows[bi][4]||'sin motivo') };
    }
  }
  if (last >= 2) {
    var rows      = sheet.getRange(2,1,last-1,9).getValues();
    var ocupados  = 0;
    var capacidad = Number(body.capacidad) || 1;
    for (var i = 0; i < rows.length; i++) {
      if (!rows[i][0]) continue;
      var f = (rows[i][0] instanceof Date) ? Utilities.formatDate(rows[i][0], 'GMT-5', 'yyyy-MM-dd') : String(rows[i][0]).trim().substring(0,10);
      if (f === body.fecha && horaToStr_(rows[i][1]) === body.horaInicio &&
          String(rows[i][2]) === body.area && String(rows[i][3]) === body.camilla) {
        ocupados++;
      }
    }
    if (ocupados >= capacidad)
      return { error: 'Espacio lleno para ese horario (' + ocupados + '/' + capacidad + ' lugares ocupados)' };
  }
  var ts = Utilities.formatDate(new Date(), 'GMT-5', 'dd/MM/yyyy HH:mm:ss');
  sheet.appendRow([body.fecha, body.horaInicio, body.area, body.camilla,
    body.paciente || '', body.docente || quien, body.observacion || '', quien, ts]);
  SpreadsheetApp.flush();
  return { ok: true };
}

function eliminarReserva(body, user) {
  var sheet = getAgendaSheet();
  var last  = sheet.getLastRow();
  if (last < 2) return { error: 'Sin reservas' };
  var rows  = sheet.getRange(2,1,last-1,9).getValues();
  for (var i = 0; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    var f = (rows[i][0] instanceof Date) ? Utilities.formatDate(rows[i][0], 'GMT-5', 'yyyy-MM-dd') : String(rows[i][0]).trim().substring(0,10);
    if (f === body.fecha && horaToStr_(rows[i][1]) === body.horaInicio &&
        String(rows[i][2]) === body.area && String(rows[i][3]) === body.camilla) {
      if (user && user.rol === 'estudiante')
        return { error: 'No tienes permiso para eliminar reservas' };
      sheet.deleteRow(i + 2);
      SpreadsheetApp.flush();
      return { ok: true };
    }
  }
  return { error: 'Reserva no encontrada' };
}

function leerBloqueos(fechaInicio, fechaFin) {
  var sheet = getBloqueoSheet();
  var last  = sheet.getLastRow();
  if (last < 2) return { ok: true, bloqueos: [] };
  var rows = sheet.getRange(2,1,last-1,7).getValues();
  var bloqueos = rows.filter(function(r){
    if (!r[0]) return false;
    var fechaStr = String(r[0]).trim();
    if (fechaStr === '*') return true;
    var parts = fechaStr.includes('/') ? fechaStr.split('/') : null;
    var fIni, fFin;
    if (parts) {
      fIni = parts[0].substring(0,10);
      fFin = parts[1].substring(0,10);
    } else {
      fIni = (r[0] instanceof Date) ? Utilities.formatDate(r[0],'GMT-5','yyyy-MM-dd') : fechaStr.substring(0,10);
      fFin = fIni;
    }
    return (!fechaInicio||fFin>=fechaInicio)&&(!fechaFin||fIni<=fechaFin);
  }).map(function(r,i){
    var fechaStr = String(r[0]).trim();
    var fechaVal;
    if (fechaStr === '*') {
      fechaVal = '*';
    } else if (fechaStr.includes('/')) {
      fechaVal = fechaStr;
    } else {
      fechaVal = (r[0] instanceof Date) ? Utilities.formatDate(r[0],'GMT-5','yyyy-MM-dd') : fechaStr.substring(0,10);
    }
    return {
      id:        i+2,
      fecha:     fechaVal,
      hora:      String(r[1]).trim(),
      area:      String(r[2]).trim(),
      camilla:   String(r[3]).trim(),
      motivo:    String(r[4]).trim(),
      creadoPor: String(r[5]).trim(),
      ts:        String(r[6]).trim()
    };
  });
  return { ok: true, bloqueos: bloqueos };
}

        asignadoPor:     String(rows[i][3] || ''),

function crearCasilleros_(codigos) {
  if (!codigos || !codigos.length) return { ok: false, error: 'Sin codigos' };
  var h        = getHojaCasilleros_();
  var rows     = getCasData_();
  var existing = {};
  rows.forEach(function(r){ existing[String(r[0]).toUpperCase()] = true; });
  var creados = 0, omitidos = 0;
  codigos.forEach(function(cod) {
    var key = String(cod).toUpperCase().trim();
    if (!key) return;
    if (existing[key]) { omitidos++; return; }
    h.appendRow([String(cod).trim(), '', '', '', '']);
    existing[key] = true;
    creados++;
  });
  return { ok: true, creados: creados, omitidos: omitidos };
}

      modificados++;
    }
  }
  try {
    var cfg = JSON.parse(sheet.getRange(1, 1).getValue());
    cfg.pctExtra = pct;
    sheet.getRange(1, 1).setValue(JSON.stringify(cfg));
  } catch(e) {}
  SpreadsheetApp.flush();

// AGREGAR PACIENTE EXTRA POR NO MEJORÍA
// ════════════════════════════════════════════════════════
function agregarPacienteExtra_(body) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(body.hoja);
  if (!sheet) return { ok: false, error: 'Hoja no encontrada' };
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  var col1    = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
  var colProm      = lastCol - 1;
  var colNotaFinal = lastCol;
  var maxCols      = lastCol - 4;
  var enAl = false, enCu = false, enSgr = false;
  var filaUltimoPac = -1, filaPacIni = -1;
  var numPacNormal = 0, numPacX = 0;
  var notasBase = 0, notasExtra = 0;
  for (var ri = 0; ri < col1.length; ri++) {
    var c0 = String(col1[ri][0]).trim();
    if (c0 === 'ALUMNO:' + body.alumno)                                                        { enAl=true; enCu=false; enSgr=false; continue; }
    if (enAl && c0.indexOf('ALUMNO:') === 0)                                                   { break; }
    if (enAl && c0 === 'CURSO:' + body.curso)                                                  { enCu=true; enSgr=false; continue; }
    if (enAl && enCu && c0.indexOf('CURSO:') === 0)                                            { enCu=false; continue; }
    if (enAl && enCu && (c0 === 'SGR:'+body.subgrupo || c0.indexOf('SGR:'+body.subgrupo+'|')===0)) {
      var partes = c0.slice(4).split('|');
      notasBase  = Number(partes[1]) || 0;
      notasExtra = Number(partes[2]) || 0;
      enSgr = true; numPacNormal = 0; numPacX = 0; filaPacIni = -1; filaUltimoPac = -1;
      continue;
    }
    if (enAl && enCu && enSgr) {
      if (c0.indexOf('SGR:') === 0) break;
      if (c0.indexOf('PACX') === 0) { numPacX++; filaUltimoPac=ri+3; if(filaPacIni===-1)filaPacIni=ri+3; }
      else if (c0.indexOf('PAC') === 0) { numPacNormal++; filaUltimoPac=ri+3; if(filaPacIni===-1)filaPacIni=ri+3; }
      else break;
    }
  }
  if (filaUltimoPac === -1 || filaPacIni === -1)
    return { ok: false, error: 'Subgrupo no encontrado: ' + body.subgrupo };
  var totalSS    = notasBase + notasExtra;
  var filaNueva  = filaUltimoPac + 1;
  var labelNuevo = 'PACX' + (numPacX + 1) + (body.pacVinculado ? '(' + body.pacVinculado + ')' : '');
  sheet.insertRowAfter(filaUltimoPac);
  sheet.getRange(filaNueva, 1).setValue(labelNuevo);
  sheet.getRange(filaNueva, 2).setValue('');
  if (totalSS > 0)
    sheet.getRange(filaNueva, 3, 1, totalSS).setValue('').setHorizontalAlignment('center');
  var sobra = maxCols - totalSS;
  if (sobra > 0) sheet.getRange(filaNueva, 3 + totalSS, 1, sobra).setBackground('#e8edf8');

  // Fórmula PROM del PACX: divisor dinámico
  var rngTodoPACX = colLetra_(3) + filaNueva + ':' + colLetra_(2 + totalSS) + filaNueva;
  var fPromPACX = '=IFERROR(ROUND(IF(COUNTIF(' + rngTodoPACX + ',"<>")>0,SUM(' + rngTodoPACX + ')/COUNTIF(' + rngTodoPACX + ',"<>"),0),2),0)';
  sheet.getRange(filaNueva, colProm).setFormula(fPromPACX);

  // PAC vinculado: su promSS = 0
  var filaVinculado = filaUltimoPac;
  if (body.pacVinculado) {
    var fv = buscarFila_(sheet, body.alumno, body.curso, body.subgrupo, body.pacVinculado);
    if (fv !== -1) filaVinculado = fv;
  }
  sheet.getRange(filaVinculado, colProm).setValue(0);

  // NOTA FINAL: SUM(todos) / totalPACs_normales
  var filaFinNueva = filaNueva; // ahora el último es el PACX recién insertado
  var rngTodos = colLetra_(colProm) + filaPacIni + ':' + colLetra_(colProm) + filaFinNueva;
  var fFinal = '=IFERROR(ROUND(SUM(' + rngTodos + ')/' + numPacNormal + ',2),0)';

  // Aplicar NOTA FINAL a todos (PAC normales + PACX)
  for (var pi = filaPacIni; pi <= filaFinNueva; pi++) {
    sheet.getRange(pi, colNotaFinal).setFormula(fFinal);
  }
  sheet.getRange(filaNueva, 1, 1, 2).setBackground('#fef3c7').setFontColor('#92400e').setFontSize(10).setFontWeight('bold');
  if (totalSS > 0) sheet.getRange(filaNueva, 3, 1, totalSS).setBackground('#fef3c7').setFontColor('#92400e').setHorizontalAlignment('center');
  if (sobra > 0)   sheet.getRange(filaNueva, 3 + totalSS, 1, sobra).setBackground('#e8edf8');
  sheet.getRange(filaNueva, colProm).setBackground('#0c4a6e').setFontColor('#7dd3fc').setFontWeight('bold').setHorizontalAlignment('center');
  sheet.getRange(filaNueva, colNotaFinal).setBackground('#1e3a8a').setFontColor('#fff').setFontWeight('bold').setHorizontalAlignment('center');
  sheet.setRowHeight(filaNueva, 22);
  SpreadsheetApp.flush();
  return { ok: true, labelNuevo: labelNuevo, fila: filaNueva };
}

// ACTUALIZAR FÓRMULAS EN TODAS LAS PLANILLAS EXISTENTES
// ════════════════════════════════════════════════════════
function actualizarTodasLasPlanillas_() {
  var ss      = SpreadsheetApp.getActiveSpreadsheet();
  var hojas   = ss.getSheets().filter(function(s){ return s.getName().indexOf('_') !== 0; });
  var actualizadas = 0, errores = [];
  hojas.forEach(function(sheet) {
    try {
      var lastRow = sheet.getLastRow();
      var lastCol = sheet.getLastColumn();
      if (lastRow < 3 || lastCol < 4) return;
      var colProm      = lastCol - 1;
      var colNotaFinal = lastCol;
      var col1         = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
      var maxCols      = lastCol - 4;
      var enAl=false, enCu=false, enSgr=false;
      var notasBase=0, notasExtra=0, pacFilas=[];
      for (var ri = 0; ri < col1.length; ri++) {
        var c0 = String(col1[ri][0]).trim();
        var filaReal = ri + 3;
        if (c0.indexOf('ALUMNO:') === 0) {
          enAl=true; enCu=false; enSgr=false;
          _flushSubgrupo(sheet,pacFilas,notasBase,notasExtra,maxCols,colProm,colNotaFinal);
          pacFilas=[];
        } else if (enAl && c0.indexOf('CURSO:') === 0) {
          enCu=true; enSgr=false;
          _flushSubgrupo(sheet,pacFilas,notasBase,notasExtra,maxCols,colProm,colNotaFinal);
          pacFilas=[];
        } else if (enAl && enCu && c0.indexOf('SGR:') === 0) {
          _flushSubgrupo(sheet,pacFilas,notasBase,notasExtra,maxCols,colProm,colNotaFinal);
          pacFilas=[];
          var partes = c0.slice(4).split('|');
          notasBase  = Number(partes[1]) || 0;
          notasExtra = Number(partes[2]) || 0;
          enSgr=true;
          var totalSS = notasBase + notasExtra;
          for (var h = 1; h <= totalSS; h++) {
            sheet.getRange(filaReal, 2+h).setValue(h===1?'EV':('S'+(h-1)));
          }
          sheet.getRange(filaReal, colProm).setValue('PROM');
          sheet.getRange(filaReal, colNotaFinal).setValue('NOTA FINAL');
        } else if (enAl && enCu && enSgr && c0.indexOf('PAC') === 0) {
          pacFilas.push({fila: filaReal, esExtra: c0.indexOf('PACX') === 0});
        }
      }
      _flushSubgrupo(sheet,pacFilas,notasBase,notasExtra,maxCols,colProm,colNotaFinal);
      SpreadsheetApp.flush();
      actualizadas++;
    } catch(ex) {
      errores.push(sheet.getName() + ': ' + ex.message);
    }
  });
  return { ok: true, actualizadas: actualizadas, errores: errores };
}

function _flushSubgrupo(sheet, pacFilas, notasBase, notasExtra, maxCols, colProm, colNotaFinal) {
  if (!pacFilas || !pacFilas.length) return;
  var totalSS    = notasBase + notasExtra;
  var filaPacIni = pacFilas[0].fila;
  var filaPacFin = pacFilas[pacFilas.length-1].fila;
  pacFilas.forEach(function(p) {
    var rowNum  = p.fila;
    var esExtra = p.esExtra;
    var rngSS    = colLetra_(3) + rowNum + ':' + colLetra_(2 + notasBase) + rowNum;
    var rngExtra = notasExtra > 0 ? colLetra_(3 + notasBase) + rowNum + ':' + colLetra_(2 + totalSS) + rowNum : null;
    var fProm;
    if (esExtra) {
      var rngTodo = colLetra_(3) + rowNum + ':' + colLetra_(2 + totalSS) + rowNum;
      fProm = '=IFERROR(ROUND(AVERAGEIF(' + rngTodo + ',"<>"),2),"")';
    } else if (notasExtra > 0) {
      var fPromSS    = 'SUM(' + rngSS + ')/' + notasBase;
      var fPromExtra = 'SUM(' + rngExtra + ')/' + notasExtra;
      fProm = '=IFERROR(ROUND(IF(COUNTIF(' + rngExtra + ',"<>")>0,(' + fPromSS + '+' + fPromExtra + ')/2,' + fPromSS + '),2),"")';
    } else {
      fProm = '=IFERROR(ROUND(SUM(' + rngSS + ')/' + notasBase + ',2),"")';
    }
    sheet.getRange(rowNum, colProm).setFormula(fProm);
    var rngPromAll = colLetra_(colProm) + filaPacIni + ':' + colLetra_(colProm) + filaPacFin;
    var fFinal = '=IFERROR(ROUND(AVERAGEIF(' + rngPromAll + ',"<>"),2),"")';
    sheet.getRange(rowNum, colNotaFinal).setFormula(fFinal);
  });
}

// AGREGAR ALUMNO A HOJA EXISTENTE
// ════════════════════════════════════════════════════════
function agregarAlumnoAHoja_(body) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(body.hoja);
  if (!sheet) return { ok: false, error: 'Hoja no encontrada' };
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  var config = {};
  try { config = JSON.parse(sheet.getRange(1,1).getValue()); } catch(e) {}
  var col1 = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
  var cursos = [], curCu = null, enAlumno = false;
  for (var ri = 0; ri < col1.length; ri++) {
    var c0 = String(col1[ri][0]).trim();
    if (c0.indexOf('ALUMNO:') === 0) {
      if (enAlumno && cursos.length > 0) break;
      enAlumno = true; curCu = null;
    } else if (enAlumno && c0.indexOf('CURSO:') === 0) {
      curCu = { nombre: c0.slice(6).trim(), subgrupos: [] };
      cursos.push(curCu);
    } else if (enAlumno && curCu && c0.indexOf('SGR:') === 0) {
      var p = c0.slice(4).split('|');
      curCu.subgrupos.push({ nombre: p[0].trim(), notasBase: Number(p[1])||0, notasExtra: Number(p[2])||0, pctExtra: Number(p[3])||0, numPacientes: 1 });
    }
  }
  if (!cursos.length) return { ok: false, error: 'No se pudo leer la estructura de la hoja' };
  var cursosSeleccionados = body.cursos || null;
  if (cursosSeleccionados && cursosSeleccionados.length) {
    cursos = cursos.filter(function(c){ return cursosSeleccionados.indexOf(c.nombre) >= 0; });
  }
  var maxCols      = lastCol - 4;
  var colProm      = lastCol - 1;
  var colNotaFinal = lastCol;
  var filaInsercion = lastRow;
  for (var ri2 = col1.length - 1; ri2 >= 0; ri2--) {
    if (String(col1[ri2][0]).trim() === '') { filaInsercion = ri2 + 3; break; }
  }
  var fila = filaInsercion;
  sheet.getRange(fila, 1).setValue('ALUMNO:' + body.nombre);
  sheet.getRange(fila, 2).setValue(body.codigo || '');
  sheet.getRange(fila, 1, 1, lastCol).setBackground('#0f172a').setFontColor('#ffffff').setFontWeight('bold').setFontSize(11);
  sheet.setRowHeight(fila, 28);
  fila++;
  for (var ci = 0; ci < cursos.length; ci++) {
    var cu = cursos[ci];
    sheet.getRange(fila, 1).setValue('CURSO:' + cu.nombre);
    sheet.getRange(fila, 1, 1, lastCol).setBackground('#1e3a5f').setFontColor('#bfdbfe').setFontWeight('bold').setFontSize(10).setFontStyle('italic');
    sheet.setRowHeight(fila, 24);
    fila++;
    for (var si = 0; si < cu.subgrupos.length; si++) {
      var sgr        = cu.subgrupos[si];
      var notasBase  = sgr.notasBase;
      var notasExtra = sgr.notasExtra;
      var totalSS    = notasBase + notasExtra;
      sheet.getRange(fila, 1).setValue('SGR:' + sgr.nombre + '|' + notasBase + '|' + notasExtra + '|' + sgr.pctExtra);
      sheet.getRange(fila, 1, 1, lastCol).setBackground('#1d4ed8').setFontColor('#e0f2fe').setFontWeight('bold').setFontSize(10);
      for (var h = 1; h <= totalSS; h++) {
        sheet.getRange(fila, 2 + h).setValue(h === 1 ? 'EV' : 'S' + (h-1));
      }
      sheet.getRange(fila, colProm).setValue('PROM');
      sheet.getRange(fila, colNotaFinal).setValue('NOTA FINAL');
      sheet.setRowHeight(fila, 22);
      fila++;
      var numPac     = Math.max(1, sgr.numPacientes || 1);
      var filaPacIni = fila;
      var filaPacFin = fila + numPac - 1;
      for (var pi = 1; pi <= numPac; pi++) {
        escribirFilaPac_(sheet, fila, pi, false, notasBase, notasExtra, maxCols, colProm, colNotaFinal, filaPacIni, filaPacFin, numPac);
        fila++;
      }
    }
  }
  sheet.getRange(fila, 1, 1, lastCol).setBackground('#1e3a5f');
  sheet.setRowHeight(fila, 5);
  SpreadsheetApp.flush();
  return { ok: true, filaInsercion: filaInsercion };
}

// ELIMINAR PACIENTE EXTRA (PACX)
// ════════════════════════════════════════════════════════
function eliminarPacX_(body) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(body.hoja);
  if (!sheet) return { ok: false, error: 'Hoja no encontrada' };
  var fila = buscarFila_(sheet, body.alumno, body.curso, body.subgrupo, body.pacLabel);
  if (fila === -1) return { ok: false, error: 'PACX no encontrado' };

  // Antes de borrar, encontrar el PAC original que abrió este PACX
  // El label del PACX es como "PACX1(PAC3)" → el PAC original es "PAC3"
  var pacLabel = String(body.pacLabel).trim();
  var pacOrigLabel = '';
  var match = pacLabel.match(/PACX\d+\((.+)\)/);
  if (match) pacOrigLabel = match[1];

  // Borrar la fila del PACX
  sheet.deleteRow(fila);
  SpreadsheetApp.flush();

  // Restaurar fórmula del PAC original si existe
  if (pacOrigLabel) {
    var filaOrig = buscarFila_(sheet, body.alumno, body.curso, body.subgrupo, pacOrigLabel);
    if (filaOrig !== -1) {
      var lastCol  = sheet.getLastColumn();
      var colProm  = lastCol - 1;
      var colNotaFinal = lastCol;

      // Leer el SGR para obtener notasBase y notasExtra
      var lastRow = sheet.getLastRow();
      var col1 = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
      var notasBase = 0, notasExtra = 0;
      var pacFilas = [], filaSgr = -1;
      var enAl = false, enCu = false, enSgr = false;

      for (var ri = 0; ri < col1.length; ri++) {
        var c0 = String(col1[ri][0]).trim();
        var fr = ri + 3;
        if (c0 === 'ALUMNO:' + body.alumno) { enAl=true; enCu=false; enSgr=false; continue; }
        if (enAl && c0.indexOf('ALUMNO:') === 0) break;
        if (enAl && c0 === 'CURSO:' + body.curso) { enCu=true; enSgr=false; continue; }
        if (enAl && enCu && c0.indexOf('CURSO:') === 0) { enCu=false; enSgr=false; continue; }
        if (enAl && enCu && c0.indexOf('SGR:') === 0) {
          var sgrNom = c0.slice(4).split('|')[0].trim().toLowerCase();
          if (sgrNom === String(body.subgrupo).trim().toLowerCase()) {
            var pp = c0.slice(4).split('|');
            notasBase  = Number(pp[1]) || 0;
            notasExtra = Number(pp[2]) || 0;
            filaSgr = fr; enSgr = true; pacFilas = []; continue;
          }
        }
        if (enAl && enCu && enSgr) {
          if (c0.indexOf('SGR:')===0||c0.indexOf('CURSO:')===0||c0.indexOf('ALUMNO:')===0) break;
          if (c0.indexOf('PAC') === 0) pacFilas.push({ fila: fr, esExtra: c0.indexOf('PACX')===0, label: c0 });
        }
      }

      // Restaurar fórmula PROM del PAC original
      var totalSS  = notasBase + notasExtra;
      var rngSS    = colLetra_(3) + filaOrig + ':' + colLetra_(2 + notasBase) + filaOrig;
      var rngExtra = notasExtra > 0 ? colLetra_(3 + notasBase) + filaOrig + ':' + colLetra_(2 + totalSS) + filaOrig : null;
      var rngTodo  = colLetra_(3) + filaOrig + ':' + colLetra_(2 + totalSS) + filaOrig;
      var fProm;
      if (notasExtra > 0) {
        fProm = '=IFERROR(ROUND(IF(COUNTIF(' + rngExtra + ',"<>")>0,' +
                'SUM(' + rngTodo + ')/COUNTIF(' + rngTodo + ',"<>"),' +
                'SUM(' + rngSS + ')/' + notasBase + '),2),0)';
      } else {
        fProm = '=IFERROR(ROUND(SUM(' + rngSS + ')/' + notasBase + ',2),0)';
      }
      sheet.getRange(filaOrig, colProm).setFormula(fProm);

      // Recalcular NOTA FINAL del subgrupo (ya no hay PACX)
      if (pacFilas.length > 0) {
        var pacNorm = pacFilas.filter(function(p){ return !p.esExtra; });
        if (pacNorm.length > 0) {
          var rngFinal = colLetra_(colProm) + pacNorm[0].fila + ':' + colLetra_(colProm) + pacNorm[pacNorm.length-1].fila;
          var fFinal = '=IFERROR(ROUND(AVERAGE(' + rngFinal + '),2),0)';
          pacNorm.forEach(function(p) {
            sheet.getRange(p.fila, colNotaFinal).setFormula(fFinal);
          });
        }
      }
      SpreadsheetApp.flush();
    }
  }

  return { ok: true };
}

// ACTUALIZAR SUBGRUPO (individual por alumno)
// ════════════════════════════════════════════════════════
function actualizarSubgrupo_(body) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(body.hoja);
  if (!sheet) return { ok: false, error: 'Hoja no encontrada' };
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  var col1    = sheet.getRange(3, 1, lastRow - 2, 1).getValues();
  var colProm      = lastCol - 1;
  var colNotaFinal = lastCol;
  var maxCols      = lastCol - 4;
  var enAl = false, enCu = false, enSgr = false;
  var filaSgr = -1, pacFilas = [];
  for (var ri = 0; ri < col1.length; ri++) {
    var c0 = String(col1[ri][0]).trim();
    var fr = ri + 3;
    if (c0 === 'ALUMNO:' + body.alumno)               { enAl=true; enCu=false; enSgr=false; pacFilas=[]; continue; }
    if (enAl && c0.indexOf('ALUMNO:') === 0)          { break; }
    if (enAl && c0 === 'CURSO:' + body.curso)          { enCu=true; enSgr=false; pacFilas=[]; continue; }
    if (enAl && enCu && c0.indexOf('CURSO:') === 0)   { enCu=false; enSgr=false; continue; }
    if (enAl && enCu && c0.indexOf('SGR:') === 0) {
      var _sgrBuscar = String(body.subgrupo).trim().toLowerCase().normalize('NFC');
      var _c0low = c0.slice(4).split('|')[0].trim().toLowerCase().normalize('NFC');
      if (_c0low === _sgrBuscar) { filaSgr = fr; enSgr = true; pacFilas = []; continue; }
    }
    if (enAl && enCu && enSgr) {
      if (c0.indexOf('SGR:') === 0 || c0.indexOf('CURSO:') === 0 || c0.indexOf('ALUMNO:') === 0) break;
      if (c0.indexOf('PAC') === 0) pacFilas.push({ fila: fr, esExtra: c0.indexOf('PACX') === 0 });
    }
  }
  if (filaSgr === -1) {
    var sgrEncontrados = col1.map(function(r){ return String(r[0]).trim(); }).filter(function(v){ return v.indexOf('SGR:')===0; }).join(' | ');
    return { ok: false, error: 'Subgrupo no encontrado. Buscado: "'+body.subgrupo+'" | Encontrados: '+sgrEncontrados };
  }
  var notasBaseNueva  = Number(body.notasBase);
  var notasExtraNueva = Number(body.notasExtra);
  var numPacNuevo     = Number(body.numPacientes) || 0;
  var celdaSgr = String(col1[filaSgr - 3][0]).trim();
  var pp = celdaSgr.slice(4).split('|');
  var pctActual = Number(pp[3]) || 0;
  var totalNuevo = notasBaseNueva + notasExtraNueva;
  sheet.getRange(filaSgr, 1).setValue('SGR:' + body.subgrupo + '|' + notasBaseNueva + '|' + notasExtraNueva + '|' + pctActual);
  for (var h = 1; h <= maxCols; h++) {
    if (h <= notasBaseNueva)  sheet.getRange(filaSgr, 2+h).setValue(h===1?'EV':'S'+(h-1));
    else if (h <= totalNuevo) sheet.getRange(filaSgr, 2+h).setValue('E'+(h-notasBaseNueva));
    else                      sheet.getRange(filaSgr, 2+h).setValue('');
  }
  var filaPacIni = pacFilas.length ? pacFilas[0].fila : -1;
  var filaPacFin = pacFilas.length ? pacFilas[pacFilas.length-1].fila : -1;
  pacFilas.forEach(function(p) {
    var rn = p.fila;
    var rngSS    = colLetra_(3)+rn+':'+colLetra_(2+notasBaseNueva)+rn;
    var rngExtra = notasExtraNueva>0 ? colLetra_(3+notasBaseNueva)+rn+':'+colLetra_(2+totalNuevo)+rn : null;
    var fProm;
    if (p.esExtra) {
      fProm = '=IFERROR(ROUND(AVERAGEIF('+colLetra_(3)+rn+':'+colLetra_(2+totalNuevo)+rn+',"<>"),2),"")';
    } else if (notasExtraNueva>0) {
      fProm = '=IFERROR(ROUND(IF(COUNTIF('+rngExtra+',"<>")>0,(SUM('+rngSS+')/'+notasBaseNueva+'+SUM('+rngExtra+')/'+notasExtraNueva+')/2,SUM('+rngSS+')/'+notasBaseNueva+'),2),"")';
    } else {
      fProm = '=IFERROR(ROUND(SUM('+rngSS+')/'+notasBaseNueva+',2),"")';
    }
    sheet.getRange(rn, colProm).setFormula(fProm);
    var rngAll = colLetra_(colProm)+filaPacIni+':'+colLetra_(colProm)+filaPacFin;
    sheet.getRange(rn, colNotaFinal).setFormula('=IFERROR(ROUND(AVERAGEIF('+rngAll+',"<>"),2),"")');
  });
  if (numPacNuevo > 0) {
    var pacNormales  = pacFilas.filter(function(p){ return !p.esExtra; });
    var numPacActual = pacNormales.length;
    if (numPacNuevo > numPacActual) {
      var filaUlt = pacFilas.length ? pacFilas[pacFilas.length-1].fila : filaSgr;
      for (var np = numPacActual+1; np <= numPacNuevo; np++) {
        sheet.insertRowAfter(filaUlt); filaUlt++;
        sheet.getRange(filaUlt,1).setValue('PAC'+np);
        sheet.getRange(filaUlt,2).setValue('');
        if (totalNuevo>0) sheet.getRange(filaUlt,3,1,totalNuevo).setValue('').setHorizontalAlignment('center');
        var sobra=maxCols-totalNuevo;
        if (sobra>0) sheet.getRange(filaUlt,3+totalNuevo,1,sobra).setBackground('#e8edf8');
        sheet.getRange(filaUlt,colProm).setBackground('#0c4a6e').setFontColor('#7dd3fc').setFontWeight('bold').setHorizontalAlignment('center');
        sheet.getRange(filaUlt,colNotaFinal).setBackground('#1e3a8a').setFontColor('#fff').setFontWeight('bold').setHorizontalAlignment('center');
        sheet.setRowHeight(filaUlt,22);
        pacFilas.push({fila:filaUlt,esExtra:false});
      }
    } else if (numPacNuevo < numPacActual) {
      for (var np2=pacNormales.length-1; np2>=numPacNuevo; np2--) {
        var filaElim=pacNormales[np2].fila;
        var rowVals=sheet.getRange(filaElim,3,1,notasBaseNueva+notasExtraNueva).getValues()[0];
        var tieneNotas=rowVals.some(function(v){return v!==''&&v!==null;});
        if (!tieneNotas) { sheet.deleteRow(filaElim); pacFilas=pacFilas.filter(function(p){return p.fila!==filaElim;}); }
      }
    }
    var filaPacIniF=pacFilas.length?pacFilas[0].fila:-1;
    var filaPacFinF=pacFilas.length?pacFilas[pacFilas.length-1].fila:-1;
    if (filaPacIniF>0) {
      pacFilas.forEach(function(p){
        var rn=p.fila;
        var rngSS2=colLetra_(3)+rn+':'+colLetra_(2+notasBaseNueva)+rn;
        var rngEx2=notasExtraNueva>0?colLetra_(3+notasBaseNueva)+rn+':'+colLetra_(2+totalNuevo)+rn:null;
        var fProm2;
        if (p.esExtra) fProm2='=IFERROR(ROUND(AVERAGEIF('+colLetra_(3)+rn+':'+colLetra_(2+totalNuevo)+rn+',"<>"),2),"")';
        else if (notasExtraNueva>0) fProm2='=IFERROR(ROUND(IF(COUNTIF('+rngEx2+',"<>")>0,(SUM('+rngSS2+')/'+notasBaseNueva+'+SUM('+rngEx2+')/'+notasExtraNueva+')/2,SUM('+rngSS2+')/'+notasBaseNueva+'),2),"")';
        else fProm2='=IFERROR(ROUND(SUM('+rngSS2+')/'+notasBaseNueva+',2),"")';
        sheet.getRange(rn,colProm).setFormula(fProm2);
        var rngAll2=colLetra_(colProm)+filaPacIniF+':'+colLetra_(colProm)+filaPacFinF;
        sheet.getRange(rn,colNotaFinal).setFormula('=IFERROR(ROUND(AVERAGEIF('+rngAll2+',"<>"),2),"")');
      });
    }
  }
  SpreadsheetApp.flush();
  return { ok: true };
}

          if (sobra>0) sheet.getRange(filaUlt,3+totalNuevo,1,sobra).setBackground('#e8edf8');
          sheet.getRange(filaUlt,colProm).setBackground('#0c4a6e').setFontColor('#7dd3fc').setFontWeight('bold').setHorizontalAlignment('center');
          sheet.getRange(filaUlt,colNotaFinal).setBackground('#1e3a8a').setFontColor('#fff').setFontWeight('bold').setHorizontalAlignment('center');
          sheet.setRowHeight(filaUlt,22);
          pacFilas.push({fila:filaUlt,esExtra:false});
        }
      } else if (numPacNuevo>0 && numPacNuevo<numPacActual) {
        for (var np2=pacNormales.length-1; np2>=numPacNuevo; np2--) {
          var filaElim=pacNormales[np2].fila;
          var rowVals=sheet.getRange(filaElim,3,1,notasBase+notasExtra).getValues()[0];
          if (!rowVals.some(function(v){return v!==''&&v!==null;})) {
            sheet.deleteRow(filaElim);
            pacFilas=pacFilas.filter(function(p){return p.fila!==filaElim;});
          }
        }
      }
      var filaPacIni=pacFilas.length?pacFilas[0].fila:-1;
      var filaPacFin=pacFilas.length?pacFilas[pacFilas.length-1].fila:-1;
      if (filaPacIni>0) {
        pacFilas.forEach(function(p){
          var rn=p.fila;
          var rngSS=colLetra_(3)+rn+':'+colLetra_(2+notasBase)+rn;
          var rngEx=notasExtra>0?colLetra_(3+notasBase)+rn+':'+colLetra_(2+totalNuevo)+rn:null;
          var fProm;
          if (p.esExtra) fProm='=IFERROR(ROUND(AVERAGEIF('+colLetra_(3)+rn+':'+colLetra_(2+totalNuevo)+rn+',"<>"),2),"")';
          else if (notasExtra>0) fProm='=IFERROR(ROUND(IF(COUNTIF('+rngEx+',"<>")>0,(SUM('+rngSS+')/'+notasBase+'+SUM('+rngEx+')/'+notasExtra+')/2,SUM('+rngSS+')/'+notasBase+'),2),"")';
          else fProm='=IFERROR(ROUND(SUM('+rngSS+')/'+notasBase+',2),"")';
          sheet.getRange(rn,colProm).setFormula(fProm);
          var rngAll=colLetra_(colProm)+filaPacIni+':'+colLetra_(colProm)+filaPacFin;
          sheet.getRange(rn,colNotaFinal).setFormula('=IFERROR(ROUND(AVERAGEIF('+rngAll+',"<>"),2),"")');
        });
      }
    }
  }
  SpreadsheetApp.flush();
  return { ok: true, errores: errores };
}

// ════════════════════════════════════════════════════════
// REGENERAR HOJA COMPLETA
// ════════════════════════════════════════════════════════
function regenerarHoja_(body) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(body.hoja);
  if (!sheet) return { ok: false, error: 'Hoja no encontrada' };
  var lastRow = sheet.getLastRow();
  if (lastRow < 3) return { ok: false, error: 'Hoja sin datos' };
  var config = {};
  try { config = JSON.parse(sheet.getRange(1,1).getValue()); } catch(e) {}
  var col1 = sheet.getRange(3, 1, lastRow-2, 1).getValues();
  var alumnosData = [];
  var curAl = null, curCu = null, curSgr = null;
  for (var ri = 0; ri < col1.length; ri++) {
    var c0 = String(col1[ri][0]).trim();
    if (c0.indexOf('ALUMNO:') === 0) {
      if (curAl) alumnosData.push(curAl);
      var cod = String(sheet.getRange(ri+3, 2).getValue()).trim();
      var nom = c0.slice(7).trim();
      curAl  = { nombre: nom, codigo: cod, cursos: [] };
      curCu  = null; curSgr = null;
    } else if (curAl && c0.indexOf('CURSO:') === 0) {
      curCu  = { nombre: c0.slice(6).trim(), subgrupos: [] };
      curAl.cursos.push(curCu);
      curSgr = null;
    } else if (curAl && c0.indexOf('SGR:') === 0 && curCu) {
      var p = c0.slice(4).split('|');
      curSgr = { nombre: p[0].trim(), notasBase: Number(p[1])||1, notasExtra: Number(p[2])||0,
                 pctExtra: Number(p[3])||Number(config.pctExtra)||20, numPacientes: 0 };
      curCu.subgrupos.push(curSgr);
    } else if (curAl && c0.indexOf('PAC') === 0 && c0.indexOf('PACX') !== 0 && curSgr) {
      curSgr.numPacientes++;
    }
  }
  if (curAl) alumnosData.push(curAl);
  if (!alumnosData.length) return { ok: false, error: 'No se encontraron alumnos' };
  var tieneCursos = alumnosData.some(function(al){ return al.cursos.length > 0; });
  if (!tieneCursos) return { ok: false, error: 'No se encontro estructura de cursos' };
  var alumnos = alumnosData.map(function(al) {
    return {
      nombre: al.nombre, codigo: al.codigo,
      cursos: al.cursos.map(function(cu) {
        return { nombre: cu.nombre, subgrupos: cu.subgrupos.map(function(sg) {
          return { nombre: sg.nombre, notasBase: sg.notasBase, notasExtra: sg.notasExtra,
                   pctExtra: sg.pctExtra, numPacientes: Math.max(1, sg.numPacientes) };
        })};
      })
    };
  });
  var nombreHoja = sheet.getName();
  var sheetIndex = sheet.getIndex();
  ss.deleteSheet(sheet);
  var resultado = crearPlanilla({ nombreHoja: nombreHoja, config: config, alumnos: alumnos });
  if (resultado.ok) {
    var hojaNew = ss.getSheetByName(nombreHoja);
    if (hojaNew) { ss.setActiveSheet(hojaNew); ss.moveActiveSheet(sheetIndex); }
  }
  return resultado;
}

// ════════════════════════════════════════════════════════
// HASH SHA-256
// ════════════════════════════════════════════════════════
function generarHashSHA256(input) {
  var rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input);
  var txtHash = "";
  for (var i = 0; i < rawHash.length; i++) {
    var hashVal = rawHash[i];
    if (hashVal < 0) hashVal += 256;
    if (hashVal.toString(16).length == 1) txtHash += '0';
    txtHash += hashVal.toString(16);
  }
  return txtHash;
}

/**
 * EJECUTAR ESTA FUNCIÓN UNA SOLA VEZ
 * Convierte todas las contraseñas de la hoja '_usuarios' a SHA-256
 */
function migrarContrasenasASH256() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_USUARIOS);
  if (!sheet) { Logger.log("Error: No se encontró la hoja de usuarios."); return; }
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  var range = sheet.getRange(2, 1, lastRow - 1, 2);
  var rows  = range.getValues();
  var migrados = 0;
  for (var i = 0; i < rows.length; i++) {
    var passwordActual = String(rows[i][1]).trim();
    if (!/^[a-f0-9]{64}$/.test(passwordActual)) {
      rows[i][1] = generarHashSHA256(passwordActual);
      Logger.log("Usuario " + rows[i][0] + " migrado.");
      migrados++;
    } else {
      Logger.log("Usuario " + rows[i][0] + " ya estaba hasheado. Omitiendo...");
    }
  }
  range.setValues(rows);
  SpreadsheetApp.flush();
  try {
    sincronizarUsuariosADestino_();
    SpreadsheetApp.getUi().alert('✅ Migración completada: ' + migrados + ' usuarios hasheados y sincronizados al Sheet destino.');
  } catch(e) {
    SpreadsheetApp.getUi().alert('✅ Migración local completada (' + migrados + ' usuarios), pero falló la sincronización al destino: ' + e.message);
  }
}

// CAMBIAR CONTRASEÑA
// ════════════════════════════════════════════════════════
function cambiarPassword_(b, usuarioAutenticado) {
  var passwordActual = String(b.passwordActual || '').trim();
  var passwordNueva  = String(b.passwordNueva  || '').trim();
  if (!passwordActual || !passwordNueva)
    return { error: 'Debes enviar passwordActual y passwordNueva' };
  if (passwordNueva.length < 6)
    return { error: 'La nueva contraseña debe tener al menos 6 caracteres' };
  var hashActual = generarHashSHA256(passwordActual);
  if (String(usuarioAutenticado.password) !== hashActual)
    return { error: 'La contraseña actual es incorrecta' };
  var hashNueva = generarHashSHA256(passwordNueva);
  var ss     = SpreadsheetApp.getActiveSpreadsheet();
  var sheet  = ss.getSheetByName(SHEET_USUARIOS);
  if (!sheet) return { error: 'Hoja de usuarios no encontrada' };
  var last   = sheet.getLastRow();
  if (last < 2) return { error: 'No hay usuarios registrados' };
  var needle = String(usuarioAutenticado.codigo).trim().toLowerCase();
  var col1   = sheet.getRange(2, 1, last - 1, 1).getValues();
  for (var i = 0; i < col1.length; i++) {
    if (String(col1[i][0]).trim().toLowerCase() === needle) {
      sheet.getRange(i + 2, 2).setValue(hashNueva);
      SpreadsheetApp.flush();
      try { sincronizarUsuariosADestino_(); } catch(e) { Logger.log('Sync error: ' + e.message); }
      return { ok: true, mensaje: 'Contraseña actualizada correctamente' };
    }
  }
  return { error: 'Usuario no encontrado' };
}

// ════════════════════════════════════════════════════════
// SINCRONIZAR _usuarios AL SHEET DESTINO
// ════════════════════════════════════════════════════════
function sincronizarUsuariosADestino_() {
  var ssOrigen  = SpreadsheetApp.getActiveSpreadsheet();
  var sheetOrig = ssOrigen.getSheetByName(SHEET_USUARIOS);
  if (!sheetOrig) throw new Error('Hoja _usuarios no encontrada en origen');
  var lastRow = sheetOrig.getLastRow();
  var lastCol = sheetOrig.getLastColumn();
  if (lastRow < 1) return { ok: true, mensaje: 'Hoja origen vacía' };
  var datos = sheetOrig.getRange(1, 1, lastRow, lastCol).getValues();

  var ssDestino;
  try {
    ssDestino = SpreadsheetApp.openById(SHEET_DESTINO_ID);
  } catch (e) {
    throw new Error('No se pudo abrir el Sheet destino (ID: ' + SHEET_DESTINO_ID + '). Detalle: ' + e.message);
  }

  var sheetDest = ssDestino.getSheetByName(SHEET_DESTINO_TAB);
  if (!sheetDest) sheetDest = ssDestino.insertSheet(SHEET_DESTINO_TAB);

  // Respaldar columnas extra del destino (más allá de las columnas del origen)
  // para no destruir ctmp, firma, gradoAcademico, cargo, etc.
  var extraPorCodigo = {};
  var extraHeaders   = [];
  var destLastRow    = sheetDest.getLastRow();
  var destLastCol    = sheetDest.getLastColumn();

  if (destLastRow >= 1 && destLastCol > lastCol) {
    var destData = sheetDest.getRange(1, 1, destLastRow, destLastCol).getValues();
    extraHeaders = destData[0].slice(lastCol);
    for (var i = 1; i < destData.length; i++) {
      var cod = String(destData[i][0]).toLowerCase().trim();
      if (cod) extraPorCodigo[cod] = destData[i].slice(lastCol);
    }
  }

  // gasindex concatena el grado al nombre de docentes ("Lic. Juan Pérez").
  // gashc los guarda por separado, así que separamos el prefijo al sincronizar.
  var PREFIJOS_GRADO = ['Lic.', 'Mtro.', 'Mtra.', 'Dr.', 'Dra.', 'Ing.', 'Prof.'];
  var idxGrado = extraHeaders.indexOf('gradoAcademico');

  // Construir datos fusionados: columnas del origen + columnas extra restauradas
  var mergedData = [];
  for (var r = 0; r < datos.length; r++) {
    var row = datos[r].slice();
    if (r === 0) {
      for (var h = 0; h < extraHeaders.length; h++) row.push(extraHeaders[h]);
    } else {
      var gradoExtraido = '';
      var rol    = String(row[2] || '').toLowerCase().trim();
      var nombre = String(row[3] || '').trim();
      if (rol === 'docente') {
        for (var p = 0; p < PREFIJOS_GRADO.length; p++) {
          if (nombre.indexOf(PREFIJOS_GRADO[p] + ' ') === 0) {
            gradoExtraido = PREFIJOS_GRADO[p];
            nombre = nombre.substring(PREFIJOS_GRADO[p].length + 1).trim();
            break;
          }
        }
        row[3] = nombre; // nombre sin prefijo de grado
      }
      var key   = String(row[0]).toLowerCase().trim();
      var extra = (extraPorCodigo[key] || []).slice();
      while (extra.length < extraHeaders.length) extra.push('');
      // Si gradoAcademico está vacío en destino, poblarlo con el grado extraído
      if (idxGrado !== -1 && gradoExtraido && !String(extra[idxGrado] || '').trim()) {
        extra[idxGrado] = gradoExtraido;
      }
      for (var e = 0; e < extraHeaders.length; e++) row.push(extra[e]);
    }
    mergedData.push(row);
  }

  var mergedCols = mergedData[0] ? mergedData[0].length : lastCol;
  sheetDest.clearContents();
  sheetDest.getRange(1, 1, mergedData.length, mergedCols).setValues(mergedData);
  sheetDest.getRange(1, 1, 1, mergedCols).setBackground('#0f172a').setFontColor('#ffffff').setFontWeight('bold');
  SpreadsheetApp.flush();
  Logger.log('Sincronización OK: ' + (lastRow - 1) + ' usuarios copiados, ' + extraHeaders.length + ' cols extra preservadas.');
  return { ok: true, usuariosSincronizados: lastRow - 1 };
}

// TRIGGER onEdit — hasheo automático al editar Col B
// ════════════════════════════════════════════════════════
function onEditUsuarios(e) {
  if (!e || !e.range) return;
  var sheet = e.range.getSheet();
  if (sheet.getName() !== SHEET_USUARIOS) return;
  var col = e.range.getColumn();
  var row = e.range.getRow();
  if (col !== 2 || row < 2) return;
  var valorNuevo = String(e.value || '').trim();
  if (!valorNuevo) return;
  if (/^[a-f0-9]{64}$/.test(valorNuevo)) return;
  var hashNuevo = generarHashSHA256(valorNuevo);
  sheet.getRange(row, 2).setValue(hashNuevo);
  SpreadsheetApp.flush();
  try { sincronizarUsuariosADestino_(); } catch (err) { Logger.log('Error sincronizando tras edición manual: ' + err.message); }
}

}

// ════════════════════════════════════════════════════════
// ACTUALIZAR FÓRMULAS DE PROMEDIO (SIN BORRAR DATOS)
// ════════════════════════════════════════════════════════
function actualizarFormulasPromedio_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  var actualizadas = 0;
  var errores = [];

  sheets.forEach(function(sheet) {
    var nombre = sheet.getName();
    if (nombre.indexOf('_') === 0) return;

    try {
      var lastRow = sheet.getLastRow();
      var lastCol = sheet.getLastColumn();
      if (lastRow < 3 || lastCol < 4) return;

      var colProm      = lastCol - 1;
      var colNotaFinal = lastCol;
      var col1 = sheet.getRange(3, 1, lastRow - 2, 1).getValues();

      var notasBase = 0, notasExtra = 0;
      var pacFilas = [];
      var filaSgr  = -1;

      function flushSgr() {
        if (pacFilas.length === 0) return;
        var totalSS = notasBase + notasExtra;

        var pacNorm = pacFilas.filter(function(p){ return !p.esExtra; });
        var pacX    = pacFilas.filter(function(p){ return p.esExtra; });

        var fFinal;
        if (pacX.length > 0) {
          var totalPACs = pacNorm.length;
          if (totalPACs < 1) totalPACs = 1;
          var filaIni = pacFilas[0].fila;
          var filaFin = pacFilas[pacFilas.length-1].fila;
          var rngTodos = colLetra_(colProm) + filaIni + ':' + colLetra_(colProm) + filaFin;
          fFinal = '=IFERROR(ROUND(SUM(' + rngTodos + ')/' + totalPACs + ',2),0)';
        } else if (pacNorm.length > 0) {
          var rngFinal = colLetra_(colProm) + pacNorm[0].fila + ':' + colLetra_(colProm) + pacNorm[pacNorm.length-1].fila;
          fFinal = '=IFERROR(ROUND(AVERAGE(' + rngFinal + '),2),0)';
        } else {
          return;
        }

        pacFilas.forEach(function(p) {
          var rn = p.fila;
          var rngSS    = colLetra_(3) + rn + ':' + colLetra_(2 + notasBase) + rn;
          var rngExtra = notasExtra > 0 ? colLetra_(3 + notasBase) + rn + ':' + colLetra_(2 + totalSS) + rn : null;
          var rngTodo  = colLetra_(3) + rn + ':' + colLetra_(2 + totalSS) + rn;

          var tienePropioPACX = pacX.some(function(px) {
            return px.label && px.label.indexOf('(' + p.label + ')') !== -1;
          });

          var fProm;
          if (p.esExtra) {
            fProm = '=IFERROR(ROUND(IF(COUNTIF(' + rngTodo + ',"<>")>0,SUM(' + rngTodo + ')/COUNTIF(' + rngTodo + ',"<>"),0),2),0)';
          } else if (tienePropioPACX) {
            fProm = '0';
          } else if (notasExtra > 0) {
            fProm = '=IFERROR(ROUND(IF(COUNTIF(' + rngExtra + ',"<>")>0,' +
                    'SUM(' + rngTodo + ')/COUNTIF(' + rngTodo + ',"<>"),' +
                    'SUM(' + rngSS + ')/' + notasBase + '),2),0)';
          } else {
            fProm = '=IFERROR(ROUND(SUM(' + rngSS + ')/' + notasBase + ',2),0)';
          }

          sheet.getRange(rn, colProm).setFormula(fProm);
          sheet.getRange(rn, colNotaFinal).setFormula(fFinal);
          actualizadas++;
        });
        pacFilas = [];
      }

      for (var ri = 0; ri < col1.length; ri++) {
        var c0 = String(col1[ri][0]).trim();
        var filaReal = ri + 3;

        if (c0.indexOf('ALUMNO:') === 0 || c0.indexOf('CURSO:') === 0) {
          flushSgr();
          notasBase = 0; notasExtra = 0; filaSgr = -1;
        } else if (c0.indexOf('SGR:') === 0) {
          flushSgr();
          var partes = c0.slice(4).split('|');
          notasBase  = Number(partes[1]) || 0;
          notasExtra = Number(partes[2]) || 0;
          filaSgr    = filaReal;
        } else if (c0.indexOf('PAC') === 0 && filaSgr !== -1) {
          pacFilas.push({ fila: filaReal, esExtra: c0.indexOf('PACX') === 0, label: c0 });
        }
      }
      flushSgr();
      SpreadsheetApp.flush();

    } catch(e) {
      errores.push({ hoja: nombre, error: e.message });
    }
  });

  return {
    ok: true,
    actualizadas: actualizadas,
    errores: errores,
    mensaje: 'Se actualizaron ' + actualizadas + ' fórmulas en ' + sheets.length + ' hojas'
  };
}

// ════════════════════════════════════════════════════════
// REPARAR CONFIG: Agrega maxCols a planillas existentes
// ════════════════════════════════════════════════════════
function repararConfigMaxCols_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  var reparadas = 0;
  var errores = [];

  sheets.forEach(function(sheet) {
    var nombre = sheet.getName();
    if (nombre.indexOf('_') === 0) return;

    try {
      var lastRow = sheet.getLastRow();
      var lastCol = sheet.getLastColumn();
      if (lastRow < 3 || lastCol < 4) return;

      var maxCols = lastCol - 4;
      if (maxCols < 1) return;

      var config = {};
      try { config = JSON.parse(sheet.getRange(1, 1).getValue()); } catch(e) {}

      if (config.maxCols !== maxCols) {
        config.maxCols = maxCols;
        sheet.getRange(1, 1).setValue(JSON.stringify(config));
        reparadas++;
      }
    } catch(e) {
      errores.push({ hoja: nombre, error: e.message });
    }
  });

  SpreadsheetApp.flush();
  return {
    ok: true,
    reparadas: reparadas,
    errores: errores,
    mensaje: 'Se repararon ' + reparadas + ' planillas con maxCols'
  };
}
